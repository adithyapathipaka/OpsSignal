1//! Counts fire-and-forget deliveries that have not finished, so a
2//! client can wait for them before the process exits.
3
4use std::sync::{Arc, Condvar, Mutex};
5use std::time::{Duration, Instant};
6
7#[derive(Default)]
8pub(crate) struct InFlight {
9    count: Mutex<usize>,
10    idle: Condvar,
11}
12
13/// Held by one delivery task. Dropping it, including when the task
14/// panics or the runtime cancels it, marks the delivery finished.
15pub(crate) struct Guard(Arc<InFlight>);
16
17impl InFlight {
18    pub(crate) fn start(self: &Arc<Self>) -> Guard {
19        *self.count.lock().unwrap_or_else(|e| e.into_inner()) += 1;
20        Guard(Arc::clone(self))
21    }
22
23    /// Blocks until nothing is in flight or `timeout` passes, and
24    /// returns how many deliveries are still unfinished.
25    pub(crate) fn wait(&self, timeout: Duration) -> usize {
26        let deadline = Instant::now() + timeout;
27        let mut count = self.count.lock().unwrap_or_else(|e| e.into_inner());
28        while *count > 0 {
29            let left = deadline.saturating_duration_since(Instant::now());
30            if left.is_zero() {
31                break;
32            }
33            count = self
34                .idle
35                .wait_timeout(count, left)
36                .unwrap_or_else(|e| e.into_inner())
37                .0;
38        }
39        *count
40    }
41}
42
43impl Drop for Guard {
44    fn drop(&mut self) {
45        let mut count = self.0.count.lock().unwrap_or_else(|e| e.into_inner());
46        *count -= 1;
47        if *count == 0 {
48            self.0.idle.notify_all();
49        }
50    }
51}

53#[cfg(test)]
54mod tests {
55    use super::*;
56
57    #[test]
58    fn wait_returns_at_once_when_idle() {
59        let in_flight = Arc::new(InFlight::default());
60        assert_eq!(in_flight.wait(Duration::from_secs(5)), 0);
61    }
62
63    #[test]
64    fn wait_times_out_with_the_unfinished_count() {
65        let in_flight = Arc::new(InFlight::default());
66        let _a = in_flight.start();
67        let _b = in_flight.start();
68        assert_eq!(in_flight.wait(Duration::from_millis(20)), 2);
69    }
70
71    #[test]
72    fn wait_wakes_when_the_last_guard_drops() {
73        let in_flight = Arc::new(InFlight::default());
74        let guard = in_flight.start();
75        let worker = std::thread::spawn(move || {
76            std::thread::sleep(Duration::from_millis(20));
77            drop(guard);
78        });
79        assert_eq!(in_flight.wait(Duration::from_secs(5)), 0);
80        worker.join().expect("worker");
81    }
