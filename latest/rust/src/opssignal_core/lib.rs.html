40/// The part of a client that a config reload replaces: the routing
41/// rules and the providers they name. Swapped as one unit, so a signal
42/// never sees new routes with old providers.
43pub struct Pipeline {
44    pub routing: RoutingConfig,
45    pub providers: HashMap<String, Arc<dyn Provider>>,
46}
47
48impl Pipeline {
49    /// Registers the configured providers. `console` is always present
50    /// because routing falls back to it.
51    pub fn from_config(config: &Config) -> Self {
52        let mut providers: HashMap<String, Arc<dyn Provider>> = HashMap::new();
53        providers.insert("console".to_string(), Arc::new(ConsoleProvider));
54        if let Some(slack) = &config.providers.slack {
55            providers.insert(
56                "slack".to_string(),
57                Arc::new(SlackProvider::new(slack.webhook_url.clone())),
58            );
59        }
60        if let Some(webhook) = &config.providers.webhook {
61            providers.insert(
62                "webhook".to_string(),
63                Arc::new(WebhookProvider::new(webhook.url.clone())),
64            );
65        }
66        if let Some(pagerduty) = &config.providers.pagerduty {
67            providers.insert(
68                "pagerduty".to_string(),
69                Arc::new(PagerDutyProvider::new(
70                    pagerduty.routing_key.clone(),
71                    pagerduty.url.clone(),
72                )),
73            );
74        }
75        if let Some(teams) = &config.providers.teams {
76            providers.insert(
77                "teams".to_string(),
78                Arc::new(TeamsProvider::new(teams.webhook_url.clone())),
79            );
80        }
81        if let Some(opsgenie) = &config.providers.opsgenie {
82            providers.insert(
83                "opsgenie".to_string(),
84                Arc::new(OpsgenieProvider::new(
85                    opsgenie.api_key.clone(),
86                    opsgenie.url.clone(),
87                    opsgenie.tags.clone(),
88                )),
89            );
90        }
91        if let Some(email) = &config.providers.email {
92            providers.insert(
93                "email".to_string(),
94                Arc::new(EmailProvider::new(email.clone())),
95            );
96        }
97        Pipeline {
98            routing: config.routing(),
99            providers,
100        }
101    }
102}

106/// The assembled pipeline: validate -> record -> route -> deliver ->
107/// complete. The signal is saved before delivery starts, so it survives
108/// a process exit mid-delivery as `pending`. A sink failure is logged
109/// and never stops delivery.
110///
111/// Dropping the client waits up to [`DEFAULT_FLUSH_TIMEOUT`] for
112/// fire-and-forget deliveries; call [`SignalClient::flush`] to choose
113/// the wait, or when the client lives in a static that is never dropped.
114pub struct SignalClient {
115    pub sink: Arc<dyn SignalSink>,
116    pipeline: SharedPipeline,
117    in_flight: Arc<InFlight>,
118    pub runtime: tokio::runtime::Runtime,
119}
120
121impl SignalClient {
122    /// Builds a client from a parsed `signal.yaml`: registers the
123    /// configured providers (console always), opens the sink, and
124    /// starts the runtime that delivery tasks run on. The config is
125    /// fixed for the client's lifetime; see [`SignalClient::from_source`]
126    /// for one that reloads.
127    pub fn from_config(config: &Config) -> Result<Self, SignalError> {
128        Self::with_runtime(config, build_runtime()?)
129    }
130
131    /// Loads the config from `source` and, if `opts.refresh` is set,
132    /// keeps re-reading it in the background and applies changes
133    /// without a restart. A reload that fails to fetch or parse keeps
134    /// the current config, so a bad edit can't take alerting down.
135    pub fn from_source(source: &ConfigSource, opts: &LoadOptions) -> Result<Self, SignalError> {
136        Self::from_sources(std::slice::from_ref(source), opts)
137    }
138
139    /// [`SignalClient::from_source`] for a layered config: `sources`
140    /// are merged lowest priority first, along with every file they
141    /// `extends`, and any of them changing triggers a reload.
142    pub fn from_sources(sources: &[ConfigSource], opts: &LoadOptions) -> Result<Self, SignalError> {
143        let runtime = build_runtime()?;
144        let loaded = runtime.block_on(source::load_initial(sources, opts))?;
145        let client = Self::with_runtime(&loaded.config, runtime)?;
146        if let Some(every) = opts.refresh {
147            reload::spawn(
148                &client.runtime,
149                reload::Reloader {
150                    roots: sources.to_vec(),
151                    opts: opts.clone(),
152                    every,
153                    last_texts: loaded.texts,
154                    storage: loaded.config.storage.clone(),
155                    pipeline: Arc::clone(&client.pipeline),
156                },
157            );
158        }
159        Ok(client)
160    }
161
162    /// The discovered config (`$SIGNAL_CONFIG`, then `./signal.yaml`,
163    /// then the zero-config default), reloaded on the interval in
164    /// `$SIGNAL_CONFIG_REFRESH_SECS`.
165    pub fn from_env() -> Result<Self, SignalError> {
166        let opts = LoadOptions::from_env()?;
167        match ConfigSource::discover()? {
168            Some(sources) => Self::from_sources(&sources, &opts),
169            None => Self::from_config(&Config::default()),
170        }
171    }
172
173    fn with_runtime(
174        config: &Config,
175        runtime: tokio::runtime::Runtime,
176    ) -> Result<Self, SignalError> {
177        let sink: Arc<dyn SignalSink> = match config.storage.sink {
178            SinkKind::Sqlite => Arc::new(SqliteSink::open(&config.storage.path)?),
179        };
180        Ok(SignalClient {
181            sink,
182            pipeline: Arc::new(RwLock::new(Arc::new(Pipeline::from_config(config)))),
183            in_flight: Arc::default(),
184            runtime,
185        })
186    }
187
188    /// The routing rules and providers currently in effect.
189    pub fn pipeline(&self) -> Arc<Pipeline> {
190        current(&self.pipeline)
191    }
192
193    /// Fire-and-forget. Blocks the caller only for the local sink
194    /// write, never on a provider. This is the default an Airflow
195    /// on_failure_callback should use — a flaky Slack webhook must never
196    /// add latency or failure risk to the DAG run itself.
197    pub fn notify_async(&self, input: SignalInput) {
198        if let Err(e) = signal::validate(&input) {
199            tracing::warn!("signal validation failed, dropping: {e}");
200            return;
201        }
202        let signal = Signal::from_input(input);
203        // On the caller's thread, so the signal is on disk before this
204        // returns even if the process exits right after.
205        record(self.sink.as_ref(), &signal);
206        let sink = Arc::clone(&self.sink);
207        let pipeline = self.pipeline();
208        let guard = self.in_flight.start();
209
210        self.runtime.spawn(async move {
211            let _guard = guard;
212            let result = deliver(&signal, &pipeline.routing, &pipeline.providers).await;
213            if let Err(e) = sink.complete(&signal, &result) {
214                tracing::error!("sink write failed: {e}");
215            }
216        });
217    }
218
219    /// Waits up to `timeout` for every [`SignalClient::notify_async`]
220    /// delivery started so far to finish. Returns false if some were
221    /// still running; those stay recorded as `pending`. Call it before
222    /// the process exits.
223    pub fn flush(&self, timeout: Duration) -> bool {
224        self.in_flight.wait(timeout) == 0
225    }
226
227    /// Blocking, bounded by timeout. Caller gets a definitive result.
228    /// Same pipeline as notify_async — only the blocking behavior
229    /// differs.
230    #[must_use = "check whether the signal was delivered"]
231    pub fn notify_sync(
232        &self,
233        input: SignalInput,
234        timeout: Duration,
235    ) -> Result<DeliveryResult, SignalError> {
236        signal::validate(&input)?;
237        let signal = Signal::from_input(input);
238        record(self.sink.as_ref(), &signal);
239        let pipeline = self.pipeline();
240
241        let result = self.runtime.block_on(async {
242            tokio::time::timeout(
243                timeout,
244                deliver(&signal, &pipeline.routing, &pipeline.providers),
245            )
246            .await
247            .unwrap_or_else(|_| DeliveryResult {
248                status: DeliveryStatus::Failed,
249                provider: None,
250                error: Some(format!("timed out after {timeout:?}")),
251                attempts: 0,
252            })
253        });
254
255        self.sink.complete(&signal, &result)?;
256        Ok(result)
257    }
258}
259
260impl Drop for SignalClient {
261    fn drop(&mut self) {
262        if !self.flush(DEFAULT_FLUSH_TIMEOUT) {
263            tracing::warn!(
264                "exiting with signals still being delivered; they stay recorded as pending"
265            );
266        }
267    }
268}
269
270/// A sink that can't record must not stop the alert going out;
271/// `complete` writes the whole row again afterwards.
272fn record(sink: &dyn SignalSink, signal: &Signal) {
273    if let Err(e) = sink.record(signal) {
274        tracing::error!("sink write failed, delivering anyway: {e}");
275    }
276}

278fn build_runtime() -> Result<tokio::runtime::Runtime, SignalError> {
279    tokio::runtime::Builder::new_multi_thread()
280        .worker_threads(2)
281        .thread_name("opssignal")
282        .enable_all()
283        .build()
284        .map_err(|e| SignalError::Internal(format!("failed to start runtime: {e}")))
285}

287/// A poisoned lock only means a reload panicked mid-swap; the value
288/// inside is still a complete pipeline, so keep using it.
289fn current(pipeline: &SharedPipeline) -> Arc<Pipeline> {
290    Arc::clone(&pipeline.read().unwrap_or_else(|e| e.into_inner()))
291}

293async fn deliver(
294    signal: &Signal,
295    routing: &RoutingConfig,
296    providers: &HashMap<String, Arc<dyn Provider>>,
297) -> DeliveryResult {
298    let provider_names = routing.resolve(signal);
299    let mut last_error = None;
300
301    for name in &provider_names {
302        if let Some(provider) = providers.get(name) {
303            // Basic retry: exponential backoff up to MAX_RETRY_ATTEMPTS.
304            // v0.2 roadmap item: add jitter, make configurable per-route.
305            for attempt in 1..=MAX_RETRY_ATTEMPTS {
306                match provider.send(signal).await {
307                    Ok(()) => {
308                        return DeliveryResult {
309                            status: DeliveryStatus::Delivered,
310                            provider: Some(provider.name().to_string()),
311                            error: None,
312                            attempts: attempt,
313                        };
314                    }
315                    Err(e) => {
316                        last_error = Some(e.to_string());
317                        if attempt < MAX_RETRY_ATTEMPTS {
318                            tokio::time::sleep(Duration::from_millis(
319                                INITIAL_BACKOFF_MS * 2u64.pow(attempt),
320                            ))
321                            .await;
322                        }
323                    }
324                }
325            }
326        } else {
327            tracing::warn!("route references unknown provider: {name}");
328        }
329    }
330
331    DeliveryResult {
332        status: DeliveryStatus::Failed,
333        provider: None,
334        error: last_error.or_else(|| Some("no provider delivered successfully".to_string())),
335        attempts: MAX_RETRY_ATTEMPTS,
336    }
337}

428    /// A provider that fails its first `fail_times` sends, then succeeds,
429    /// counting every call so tests can assert on retry behaviour.
430    struct FlakyProvider {
431        name: &'static str,
432        fail_times: u32,
433        calls: std::sync::atomic::AtomicU32,
434    }
435
436    impl FlakyProvider {
437        fn new(name: &'static str, fail_times: u32) -> Arc<Self> {
438            Arc::new(FlakyProvider {
439                name,
440                fail_times,
441                calls: std::sync::atomic::AtomicU32::new(0),
442            })
443        }
444
445        fn calls(&self) -> u32 {
446            self.calls.load(std::sync::atomic::Ordering::SeqCst)
447        }
448    }
449
450    #[async_trait::async_trait]
451    impl Provider for FlakyProvider {
452        fn name(&self) -> &'static str {
453            self.name
454        }
455
456        async fn send(&self, _signal: &Signal) -> Result<(), SignalError> {
457            let call = self.calls.fetch_add(1, std::sync::atomic::Ordering::SeqCst) + 1;
458            if call <= self.fail_times {
459                Err(SignalError::Delivery(format!(
460                    "{} failure #{call}",
461                    self.name
462                )))
463            } else {
464                Ok(())
465            }
466        }
467    }
