4pub struct SlackProvider {
5    pub webhook_url: String,
6    client: reqwest::Client,
7}
8
9impl SlackProvider {
10    pub fn new(webhook_url: String) -> Self {
11        SlackProvider {
12            webhook_url,
13            client: reqwest::Client::new(),
14        }
15    }
16
17    fn emoji_for(severity: Severity) -> &'static str {
18        match severity {
19            Severity::Info => ":information_source:",
20            Severity::Success => ":white_check_mark:",
21            Severity::Warning => ":warning:",
22            Severity::Error => ":x:",
23            Severity::Critical => ":rotating_light:",
24        }
25    }
26}
27
28#[async_trait::async_trait]
29impl Provider for SlackProvider {
30    fn name(&self) -> &'static str {
31        "slack"
32    }
33
34    async fn send(&self, signal: &Signal) -> Result<(), SignalError> {
35        let text = format!(
36            "{} *{}*\n{}",
37            Self::emoji_for(signal.severity),
38            signal.title,
39            signal.message.clone().unwrap_or_default()
40        );
41        let body = serde_json::json!({ "text": text });
42
43        let resp = self
44            .client
45            .post(&self.webhook_url)
46            .json(&body)
47            .send()
48            .await
49            .map_err(|e| SignalError::Delivery(e.to_string()))?;
50
51        if !resp.status().is_success() {
52            return Err(SignalError::Delivery(format!(
53                "slack webhook returned status {}",
54                resp.status()
55            )));
56        }
57        Ok(())
58    }
59}

79    #[test]
80    fn every_severity_has_a_distinct_emoji() {
81        let all = [
82            Severity::Info,
83            Severity::Success,
84            Severity::Warning,
85            Severity::Error,
86            Severity::Critical,
87        ];
88        let mut emojis: Vec<&str> = all.iter().map(|s| SlackProvider::emoji_for(*s)).collect();
89        emojis.sort_unstable();
90        emojis.dedup();
91        assert_eq!(emojis.len(), all.len());
92        assert_eq!(
93            SlackProvider::emoji_for(Severity::Critical),
94            ":rotating_light:"
95        );
96    }
