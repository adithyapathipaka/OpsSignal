10/// Covers connect, TLS, auth and the whole SMTP exchange of one attempt,
11/// so a server that hangs costs one attempt rather than the retry loop.
12const SMTP_TIMEOUT: Duration = Duration::from_secs(30);

14/// Sends a plain-text email through an SMTP server or, in a build with
15/// AWS support, the Amazon SES API. Repeats of the same
16/// signal keep the same subject and `References` header, so mail
17/// clients thread them into one conversation instead of filling the
18/// inbox.
19pub struct EmailProvider {
20    config: EmailConfig,
21}

28    fn subject_for(&self, signal: &Signal) -> String {
29        let severity = signal.severity.to_string().to_uppercase();
30        match &self.config.subject_prefix {
31            Some(prefix) if !prefix.is_empty() => {
32                format!("{prefix} [{severity}] {}", signal.title)
33            }
34            _ => format!("[{severity}] {}", signal.title),
35        }
36    }

38    fn body_for(signal: &Signal) -> String {
39        let mut body = format!("{}\n", signal.title);
40        if let Some(message) = &signal.message {
41            body.push('\n');
42            body.push_str(message);
43            body.push('\n');
44        }
45        body.push('\n');
46        let severity = signal.severity.to_string();
47        let mut fields: Vec<(&str, &str)> = vec![
48            ("Severity", &severity),
49            ("Source", &signal.source),
50            ("Event", &signal.event_type),
51        ];
52        let optional = [
53            ("Environment", &signal.environment),
54            ("Service", &signal.service),
55            ("Component", &signal.component),
56            ("Team", &signal.team),
57            ("Run ID", &signal.run_id),
58            ("Correlation ID", &signal.correlation_id),
59            ("Trace ID", &signal.trace_id),
60            ("Logs", &signal.log_url),
61        ];
62        for (label, value) in optional {
63            if let Some(value) = value {
64                fields.push((label, value));
65            }
66        }
67        for (label, value) in fields {
68            body.push_str(&format!("{:<16}{value}\n", format!("{label}:")));
69        }
70        if signal.metadata.as_object().is_some_and(|m| !m.is_empty()) {
71            let metadata = serde_json::to_string_pretty(&signal.metadata)
72                .unwrap_or_else(|_| signal.metadata.to_string());
73            body.push_str(&format!("\nMetadata:\n{metadata}\n"));
74        }
75        body.push_str(&format!(
76            "\n--\nSignal {} at {}\nSent by OpsSignal\n",
77            signal.id,
78            signal.timestamp.to_rfc3339()
79        ));
80        body
81    }

83    fn message_for(&self, signal: &Signal) -> Result<Message, SignalError> {
84        let mut builder: MessageBuilder = Message::builder()
85            .from(mailbox(&self.config.from)?)
86            .subject(self.subject_for(signal))
87            .header(ContentType::TEXT_PLAIN);
88        for to in &self.config.to {
89            builder = builder.to(mailbox(to)?);
90        }
91        for cc in &self.config.cc {
92            builder = builder.cc(mailbox(cc)?);
93        }
94        if let Some(key) = &signal.dedup_key {
95            let thread = thread_id(key);
96            builder = builder.in_reply_to(thread.clone()).references(thread);
97        }
98        builder
99            .body(Self::body_for(signal))
100            .map_err(|e| SignalError::Delivery(format!("cannot build the email: {e}")))
101    }

103    fn transport(smtp: &SmtpConfig) -> Result<AsyncSmtpTransport<Tokio1Executor>, SignalError> {
104        let builder = match smtp.tls {
105            SmtpTls::Starttls => AsyncSmtpTransport::<Tokio1Executor>::starttls_relay(&smtp.host),
106            SmtpTls::Tls => AsyncSmtpTransport::<Tokio1Executor>::relay(&smtp.host),
107            SmtpTls::None => Ok(AsyncSmtpTransport::<Tokio1Executor>::builder_dangerous(
108                &smtp.host,
109            )),
110        }
111        .map_err(|e| SignalError::Delivery(format!("smtp: {e}")))?;
112        let mut builder = builder.port(smtp.port()).timeout(Some(SMTP_TIMEOUT));
113        if let (Some(username), Some(password)) = (&smtp.username, &smtp.password) {
114            builder = builder.credentials(Credentials::new(username.clone(), password.clone()));
115        }
116        Ok(builder.build())
117    }
118}

120fn mailbox(address: &str) -> Result<Mailbox, SignalError> {
121    address
122        .parse()
123        .map_err(|e| SignalError::Delivery(format!("invalid email address {address:?}: {e}")))
124}

126/// A message id that every email for the same dedup key refers to.
127/// Hashed because caller-supplied keys may hold characters a header
128/// can't carry.
129fn thread_id(dedup_key: &str) -> String {
130    let digest = format!("{:x}", Sha256::digest(dedup_key.as_bytes()));
131    format!("<{}@opssignal>", &digest[..32])
132}

135impl Provider for EmailProvider {
136    fn name(&self) -> &'static str {
137        "email"
138    }
139
140    async fn send(&self, signal: &Signal) -> Result<(), SignalError> {
141        match self.config.transport() {
142            Some(EmailTransport::Smtp(smtp)) => {
143                let message = self.message_for(signal)?;
144                Self::transport(smtp)?
145                    .send(message)
146                    .await
147                    .map_err(|e| SignalError::Delivery(format!("smtp: {e}")))?;
148                Ok(())
149            }
150            Some(EmailTransport::Ses(ses)) => self.send_ses(ses, signal).await,
151            None => Err(SignalError::Delivery(
152                "email needs exactly one of smtp and ses".into(),
153            )),
154        }
155    }
156}

162    async fn send_ses(&self, ses: &SesConfig, signal: &Signal) -> Result<(), SignalError> {
163        let raw = self.message_for(signal)?.formatted();
164        crate::cloud::aws::send_ses_raw(
165            ses.region.as_deref(),
166            ses.configuration_set.as_deref(),
167            &raw,
168        )
169        .await
170    }

177        super::external::send_external("aws", "email", config_json, signal).await
178    }

221    /// A one-connection SMTP server that accepts everything except
222    /// recipients in `reject_rcpt`, and records what the client sent.
223    async fn fake_smtp(reject_rcpt: bool) -> (u16, Arc<Mutex<String>>) {
224        let listener = TcpListener::bind("127.0.0.1:0").await.expect("bind");
225        let port = listener.local_addr().expect("addr").port();
226        let transcript = Arc::new(Mutex::new(String::new()));
227        let log = Arc::clone(&transcript);
228        tokio::spawn(async move {
229            let Ok((socket, _)) = listener.accept().await else {
230                return;
231            };
232            let (read, mut write) = socket.into_split();
233            let mut lines = BufReader::new(read).lines();
234            let _ = write.write_all(b"220 fake ESMTP\r\n").await;
235            let mut in_data = false;
236            while let Ok(Some(line)) = lines.next_line().await {
237                log.lock().expect("log").push_str(&format!("{line}\n"));
238                let reply: &[u8] = if in_data {
239                    if line == "." {
240                        in_data = false;
241                        b"250 queued\r\n"
242                    } else {
243                        continue;
244                    }
245                } else {
246                    let verb = line.split(' ').next().unwrap_or("").to_uppercase();
247                    match verb.as_str() {
248                        "EHLO" | "HELO" => b"250-fake\r\n250 AUTH PLAIN LOGIN\r\n",
249                        "AUTH" => b"235 ok\r\n",
250                        "RCPT" if reject_rcpt => b"550 no such user\r\n",
251                        "DATA" => {
252                            in_data = true;
253                            b"354 go ahead\r\n"
254                        }
255                        "QUIT" => {
256                            let _ = write.write_all(b"221 bye\r\n").await;
257                            return;
258                        }
259                        _ => b"250 ok\r\n",
260                    }
261                };
262                let _ = write.write_all(reply).await;
263            }
264        });
265        (port, transcript)
266    }

395        struct FakeAddon(Calls);
396        impl ExternalSender for FakeAddon {
397            fn send(&self, provider: &str, config: &str, signal: &str) -> Result<(), String> {
398                self.0
399                    .lock()
400                    .expect("calls")
401                    .push((provider.into(), config.into(), signal.into()));

403            }
