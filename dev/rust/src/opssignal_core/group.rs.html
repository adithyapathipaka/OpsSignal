1//! Route `group_by` and `group_wait`: one digest instead of a burst.
2//!
3//! A grouped route does not send a matching signal at once. It adds the
4//! signal to the open group for its `group_by` values, and the first
5//! signal of a group starts its `group_wait`. When that runs out, or the
6//! client is flushed (as every binding does on the way out), the group
7//! is sent as one digest: a signal of its own, built from the members.
8//! A group of one is sent as that signal, unchanged.
9//!
10//! Groups live in the client's memory, so they collect signals sent
11//! through one client, in one process. Grouping across processes or
12//! hosts is out of scope for the embedded SDK. Every member is stored
13//! by the sink before it is grouped, so a process that dies with a
14//! group open leaves its members on record as `grouped`.
15

25/// A group this full is sent at once, and the next signal opens a new
26/// one. Bounds memory when a group_wait is long and signals are many.
27pub const MAX_GROUP_SIZE: usize = 500;
28
29/// How many members a digest's message lists before "and N more".
30const LISTED_IN_MESSAGE: usize = 20;
31
32/// How many members a digest lists in `metadata.digest.signals`.
33const LISTED_IN_METADATA: usize = 100;
34
35/// A signal held in a group.
36pub(crate) struct Member {
37    pub signal: Signal,
38    /// Its `dedup` record on this route, released if the digest fails.
39    pub admission: Admission,
40    /// Whether the member's stored row waits on the digest (`grouped`)
41    /// and should take its outcome. False when the member was also
42    /// sent at once on another route; that outcome stays.
43    pub owns_row: bool,
44}
45
46/// A group taken out of the open set, ready to send.
47pub(crate) struct Closed {
48    pub route: Route,
49    pub by: Vec<(String, Option<String>)>,
50    pub members: Vec<Member>,
51    /// Counts the send as in flight from the moment the group closes,
52    /// so a flush can't miss it.
53    pub _guard: Guard,
54}
55
56struct Open {
57    seq: u64,
58    route: Route,
59    by: Vec<(String, Option<String>)>,
60    members: Vec<Member>,
61}
62
63/// What adding a signal did.
64pub(crate) enum Added {
65    /// Joined an open group.
66    Joined,
67    /// Opened a group: the caller closes it with [`Groups::take`]
68    /// after `wait`.
69    Opened {
70        key: String,
71        seq: u64,
72        wait: Duration,
73    },
74    /// Filled a group, which must be sent now.
75    Full(Closed),
76}
77
78/// The open groups of one client.
79pub(crate) struct Groups {
80    open: Mutex<HashMap<String, Open>>,
81    next_seq: Mutex<u64>,
82    in_flight: Arc<InFlight>,
83}
84
85impl Groups {
86    pub(crate) fn new(in_flight: Arc<InFlight>) -> Self {
87        Groups {
88            open: Mutex::default(),
89            next_seq: Mutex::new(0),
90            in_flight,
91        }
92    }
93
94    /// Adds `member` to its group on `route`, opening one if needed.
95    pub(crate) fn add(&self, route: &Route, member: Member) -> Added {
96        let by = group_values(route, &member.signal);
97        let key = group_key(route, &by);
98        let mut open = self.open.lock().unwrap_or_else(|e| e.into_inner());
99        if let Some(group) = open.get_mut(&key) {
100            group.members.push(member);
101            if group.members.len() < MAX_GROUP_SIZE {
102                return Added::Joined;
103            }
104            return match open.remove(&key) {
105                Some(group) => Added::Full(self.close(group)),
106                None => Added::Joined,
107            };
108        }
109        let seq = {
110            let mut next = self.next_seq.lock().unwrap_or_else(|e| e.into_inner());
111            *next += 1;
112            *next
113        };
114        open.insert(
115            key.clone(),
116            Open {
117                seq,
118                route: route.clone(),
119                by,
120                members: vec![member],
121            },
122        );
123        Added::Opened {
124            key,
125            seq,
126            wait: route.group_window().unwrap_or_default(),
127        }
128    }
129
130    /// Closes the group `key` if it is still the one opened as `seq`
131    /// (it may have been sent full or by a flush meanwhile).
132    pub(crate) fn take(&self, key: &str, seq: u64) -> Option<Closed> {
133        let mut open = self.open.lock().unwrap_or_else(|e| e.into_inner());
134        if open.get(key)?.seq != seq {
135            return None;
136        }
137        open.remove(key).map(|group| self.close(group))
138    }
139
140    /// Closes every open group.
141    pub(crate) fn take_all(&self) -> Vec<Closed> {
142        let mut open = self.open.lock().unwrap_or_else(|e| e.into_inner());
143        open.drain().map(|(_, group)| self.close(group)).collect()
144    }
145
146    fn close(&self, group: Open) -> Closed {
147        Closed {
148            route: group.route,
149            by: group.by,
150            members: group.members,
151            _guard: self.in_flight.start(),
152        }
153    }
154}

156/// The values of a route's `group_by` fields for `signal`, in order.
157/// A field the signal doesn't set is `None`.
158pub(crate) fn group_values(route: &Route, signal: &Signal) -> Vec<(String, Option<String>)> {
159    route
160        .group_by
161        .iter()
162        .flatten()
163        .map(|field| (field.clone(), field_value(signal, field)))
164        .collect()
165}
166
167fn group_key(route: &Route, by: &[(String, Option<String>)]) -> String {
168    format!(
169        "{}\n{}",
170        crate::throttle::route_key(route),
171        serde_json::to_string(by).unwrap_or_default()
172    )
173}

175/// Whether `field` can be used in `group_by`.
176pub fn is_group_field(field: &str) -> bool {
177    GROUP_FIELDS.contains(&field)
178        || field
179            .strip_prefix("metadata.")
180            .is_some_and(|path| !path.is_empty() && path.split('.').all(|p| !p.is_empty()))
181}
182
183/// The value of a `group_by` field: a signal field, or a dotted path
184/// into its metadata. Strings are taken as they are, other JSON values
185/// as JSON text.
186pub(crate) fn field_value(signal: &Signal, field: &str) -> Option<String> {
187    if let Some(path) = field.strip_prefix("metadata.") {
188        let mut value = &signal.metadata;
189        for part in path.split('.') {
190            value = value.get(part)?;
191        }
192        return match value {
193            serde_json::Value::Null => None,
194            serde_json::Value::String(s) => Some(s.clone()),
195            other => Some(other.to_string()),
196        };
197    }
198    let text = |v: &Option<String>| v.clone();
199    match field {
200        "source" => Some(signal.source.clone()),
201        "service" => text(&signal.service),
202        "component" => text(&signal.component),
203        "event_type" => Some(signal.event_type.clone()),
204        "severity" => Some(signal.severity.to_string()),
205        "title" => Some(signal.title.clone()),
206        "environment" => text(&signal.environment),
207        "team" => text(&signal.team),
208        "run_id" => text(&signal.run_id),
209        "correlation_id" => text(&signal.correlation_id),
210        "dedup_key" => text(&signal.dedup_key),
211        _ => None,
212    }
213}

215fn rank(severity: Severity) -> u8 {
216    match severity {
217        Severity::Info => 0,
218        Severity::Success => 1,
219        Severity::Warning => 2,
220        Severity::Error => 3,
221        Severity::Critical => 4,
222    }
223}

225/// The value every member shares, or `None` if they differ or none
226/// sets it.
227fn common<F>(members: &[Member], get: F) -> Option<String>
228where
229    F: Fn(&Signal) -> Option<&String>,
230{
231    let first = get(&members.first()?.signal)?;
232    members
233        .iter()
234        .all(|m| get(&m.signal) == Some(first))
235        .then(|| first.clone())
236}

238/// Builds the digest for a group of two or more. It is a signal like
239/// any other, so templates, dedup and the sink treat it the usual way:
240///
241/// - `title`: the first member's title and how many more there are;
242/// - `message`: one line per member, the first [`LISTED_IN_MESSAGE`];
243/// - `severity`: the highest of the members;
244/// - `source`, `environment` and the other fields: the members' value
245///   when they all share one (`source` is `opssignal` otherwise);
246/// - `metadata.digest`: `count`, the `group` values, `first_at`,
247///   `last_at` and `signals` (id, title, severity, source, event_type,
248///   timestamp of each member, the first [`LISTED_IN_METADATA`]).
249pub(crate) fn digest(closed: &Closed) -> Signal {
250    let members = &closed.members;
251    let first = &members[0].signal;
252    let count = members.len();
253    let severity = members
254        .iter()
255        .map(|m| m.signal.severity)
256        .max_by_key(|s| rank(*s))
257        .unwrap_or_default();
258
259    let mut lines: Vec<String> = members
260        .iter()
261        .take(LISTED_IN_MESSAGE)
262        .map(|m| format!("- [{}] {}", m.signal.severity, m.signal.title))
263        .collect();
264    if count > LISTED_IN_MESSAGE {
265        lines.push(format!("...and {} more", count - LISTED_IN_MESSAGE));
266    }
267
268    let signals: Vec<serde_json::Value> = members
269        .iter()
270        .take(LISTED_IN_METADATA)
271        .map(|m| {
272            serde_json::json!({
273                "id": m.signal.id,
274                "title": m.signal.title,
275                "severity": m.signal.severity,
276                "source": m.signal.source,
277                "event_type": m.signal.event_type,
278                "timestamp": m.signal.timestamp,
279            })
280        })
281        .collect();
282    let group: serde_json::Map<String, serde_json::Value> = closed
283        .by
284        .iter()
285        .map(|(field, value)| (field.clone(), serde_json::json!(value)))
286        .collect();
287    let first_at = members.iter().map(|m| m.signal.timestamp).min();
288    let last_at = members.iter().map(|m| m.signal.timestamp).max();
289
290    let key = format!(
291        "digest:{:x}",
292        Sha256::digest(group_key(&closed.route, &closed.by).as_bytes())
293    );
294
295    Signal::from_input(SignalInput {
296        source: common(members, |s| Some(&s.source)).unwrap_or_else(|| "opssignal".into()),
297        service: common(members, |s| s.service.as_ref()),
298        component: common(members, |s| s.component.as_ref()),
299        event_type: common(members, |s| Some(&s.event_type)).unwrap_or_else(|| "digest".into()),
300        severity,
301        title: format!("{} (+{} more)", first.title, count - 1),
302        message: Some(lines.join("\n")),
303        environment: common(members, |s| s.environment.as_ref()),
304        team: common(members, |s| s.team.as_ref()),
305        run_id: common(members, |s| s.run_id.as_ref()),
306        correlation_id: common(members, |s| s.correlation_id.as_ref()),
307        trace_id: common(members, |s| s.trace_id.as_ref()),
308        log_url: common(members, |s| s.log_url.as_ref()),
309        metadata: Some(serde_json::json!({
310            "digest": {
311                "count": count,
312                "group": group,
313                "first_at": first_at,
314                "last_at": last_at,
315                "signals": signals,
316            }
317        })),
318        dedup_key: Some(key),
319    })
320}

424    #[test]
425    fn take_ignores_a_group_already_sent() {
426        let g = groups();
427        let r = route(&[], Some(60));
428        let Added::Opened { key, seq, wait } = g.add(&r, member(signal("a", Severity::Info, "1")))
429        else {
430            panic!("expected a new group");
431        };
432        assert_eq!(wait, Duration::from_secs(60));
433        assert_eq!(g.take_all().len(), 1);
434        // The timer fires after a flush sent the group: nothing to do,
435        // even when a new group has opened under the same key.
436        g.add(&r, member(signal("b", Severity::Info, "1")));
437        assert!(g.take(&key, seq).is_none());
438        assert_eq!(g.take_all().len(), 1);
439    }

441    #[test]
442    fn a_full_group_is_sent_at_once() {
443        let g = groups();
444        let r = route(&[], None);
445        for i in 0..MAX_GROUP_SIZE - 1 {
446            assert!(!matches!(
447                g.add(&r, member(signal(&format!("s{i}"), Severity::Info, "1"))),
448                Added::Full(_)
449            ));
450        }
451        match g.add(&r, member(signal("last", Severity::Info, "1"))) {
452            Added::Full(closed) => assert_eq!(closed.members.len(), MAX_GROUP_SIZE),
453            _ => panic!("expected the group to be full"),

458    #[test]
459    fn digest_summarises_its_members() {
460        let g = groups();
461        let r = route(&["run_id"], None);
462        g.add(&r, member(signal("test a failed", Severity::Warning, "9")));
463        g.add(&r, member(signal("test b failed", Severity::Error, "9")));
464        let mut other = signal("test c failed", Severity::Warning, "9");
465        other.environment = Some("staging".into());
466        g.add(&r, member(other));
467        let closed = g.take_all().pop().expect("group");
468
469        let d = digest(&closed);
470        assert_eq!(d.title, "test a failed (+2 more)");
471        assert_eq!(d.severity, Severity::Error);
472        assert_eq!(d.source, "dbt");
473        assert_eq!(d.run_id.as_deref(), Some("9"));
474        assert_eq!(d.environment, None, "members disagree");
475        assert_eq!(
476            d.message.as_deref(),
477            Some("- [warning] test a failed\n- [error] test b failed\n- [warning] test c failed")
478        );
479        let meta = &d.metadata["digest"];
480        assert_eq!(meta["count"], 3);
481        assert_eq!(meta["group"]["run_id"], "9");
482        assert_eq!(meta["signals"][1]["title"], "test b failed");
483        assert!(d.dedup_key.expect("key").starts_with("digest:"));
484        // Its own id, never a member's.
485        assert!(closed.members.iter().all(|m| m.signal.id != d.id));
486    }

490        let g = groups();
