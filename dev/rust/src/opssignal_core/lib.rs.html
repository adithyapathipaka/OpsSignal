148    /// What each provider `signal` routes to would be sent, in plan
149    /// order, without sending anything or recording the signal; with
150    /// `only`, just that provider whether or not a route picks it. A
151    /// provider that can't show its payload has `None`.
152    pub fn preview(
153        &self,
154        signal: &Signal,
155        only: Option<&str>,
156    ) -> Result<Vec<(String, Option<serde_json::Value>)>, SignalError> {
157        let names: Vec<String> = match only {
158            Some(name) => vec![name.to_string()],
159            None => {
160                // A grouped route's providers would get a digest later;
161                // show them what this signal looks like on its own.
162                let plan = self.routing.plan(signal);
163                let mut names: Vec<String> = plan.providers().cloned().collect();
164                for (index, _) in &plan.grouped {
165                    for name in &self.routing.routes[*index].providers {
166                        if !names.contains(name) {
167                            names.push(name.clone());
168                        }
169                    }
170                }
171                names
172            }
173        };
174        names
175            .into_iter()
176            .map(|name| match self.providers.get(&name) {
177                Some(provider) => {
178                    let payload = provider.preview(signal);
179                    Ok((name, payload))
180                }
181                None => {
182                    let mut known: Vec<&String> = self.providers.keys().collect();
183                    known.sort();
184                    let known: Vec<&str> = known.iter().map(|s| s.as_str()).collect();
185                    Err(SignalError::Config(format!(
186                        "provider '{name}' is not configured (configured: {})",
187                        known.join(", ")
188                    )))
189                }
190            })
191            .collect()
192    }

197/// The assembled pipeline: validate -> record -> route -> deliver ->
198/// complete. The signal is saved before delivery starts, so it survives
199/// a process exit mid-delivery as `pending`, and each provider's
200/// outcome is saved as soon as it is known. A sink failure is logged
201/// and never stops delivery.
202///
203/// Dropping the client waits up to [`DEFAULT_FLUSH_TIMEOUT`] for
204/// fire-and-forget deliveries; call [`SignalClient::flush`] to choose
205/// the wait, or when the client lives in a static that is never dropped.
206pub struct SignalClient {
207    pub sink: Arc<dyn SignalSink>,
208    pipeline: SharedPipeline,
209    /// Route `dedup` and `rate_limit` history. `None` if its table
210    /// could not be opened; routes then send without throttling.
211    throttle: Option<Arc<Throttle>>,
212    /// Which providers hold an open alert for each dedup key. `None` if
213    /// its table could not be opened; an `acknowledge` or `resolve` then
214    /// goes only to providers that resolve by key.
215    alerts: Option<Arc<AlertBook>>,
216    in_flight: Arc<InFlight>,
217    /// Signals held by routes that set `group_by` or `group_wait`.
218    groups: Arc<Groups>,
219    pub runtime: tokio::runtime::Runtime,
220}

240    /// [`SignalClient::from_source`] for a layered config: `sources`
241    /// are merged lowest priority first, along with every file they
242    /// `extends`, and any of them changing triggers a reload.
243    pub fn from_sources(sources: &[ConfigSource], opts: &LoadOptions) -> Result<Self, SignalError> {
244        let runtime = build_runtime()?;

254                    last_texts: loaded.texts,
255                    storage: loaded.config.storage.clone(),
256                    pipeline: Arc::clone(&client.pipeline),

259        }
260        Ok(client)
261    }

314    /// Fire-and-forget. Blocks the caller only for the local sink
315    /// write, never on a provider. This is the default an Airflow
316    /// on_failure_callback should use — a flaky Slack webhook must never
317    /// add latency or failure risk to the DAG run itself.
318    pub fn notify_async(&self, input: SignalInput) {
319        if let Err(e) = signal::validate(&input) {
320            tracing::warn!("signal validation failed, dropping: {e}");
321            return;
322        }
323        let pipeline = self.pipeline();
324        // Also on the caller's thread, so dedup sees signals in the
325        // order they were sent.
326        let Prepared {
327            signal,
328            plan,
329            mut admission,
330            prior_refs,
331        } = self.prepare(input, &pipeline);
332        // On the caller's thread, so the signal is on disk before this
333        // returns even if the process exits right after.
334        record(self.sink.as_ref(), &signal);
335        let sink = Arc::clone(&self.sink);
336        if plan.targets.is_empty() {
337            self.settle_unsent(&pipeline, &plan, signal, &mut admission);
338            return;
339        }
340        self.hold(&pipeline, &plan, &signal, &mut admission, false);
341        let throttle = self.throttle.clone();
342        let alerts = self.alerts.clone();
343        let guard = self.in_flight.start();
344
345        self.runtime.spawn(async move {
346            let _guard = guard;
347            let tracker = Tracker::new(&signal, sink.as_ref(), prior_refs);
348            let result = deliver(&tracker, &plan, &pipeline.providers).await;
349            settle(throttle.as_deref(), &admission, &result);
350            if let Some(alerts) = &alerts {
351                alerts.record(&signal, &result, &tracker.refs());
352            }
353            if let Err(e) = sink.complete(&signal, &result) {
354                tracing::error!("sink write failed: {e}");
355            }
356        });
357    }

359    /// Sends every open group now, then waits up to `timeout` for every
360    /// [`SignalClient::notify_async`] delivery and digest started so
361    /// far to finish. Returns false if some were still running; those
362    /// stay recorded as `pending` (or `grouped`). Call it before the
363    /// process exits.
364    pub fn flush(&self, timeout: Duration) -> bool {
365        for closed in self.groups.take_all() {
366            self.spawn_group(closed);
367        }
368        self.in_flight.wait(timeout) == 0
369    }

371    /// Sends every open group now and waits up to `timeout` for them,
372    /// returning the outcome of each group that finished: the digest's,
373    /// or the signal's own for a group of one. For a caller that needs
374    /// to know, such as the CLI's exit code. Groups still sending when
375    /// the wait runs out carry on, and [`SignalClient::flush`] waits
376    /// for them.
377    pub fn flush_groups(&self, timeout: Duration) -> Vec<DeliveryResult> {
378        let handles: Vec<_> = self
379            .groups
380            .take_all()
381            .into_iter()
382            .map(|closed| self.spawn_group(closed))
383            .collect();
384        self.runtime.block_on(async {
385            let deadline = tokio::time::Instant::now() + timeout;
386            let mut results = Vec::new();
387            for handle in handles {
388                if let Ok(Ok(result)) = tokio::time::timeout_at(deadline, handle).await {
389                    results.push(result);
390                }
391            }
392            results
393        })
394    }

396    /// Stores the outcome of a signal no provider gets now: every
397    /// matching route held it back, or holds it for a digest. Grouping
398    /// comes after the row is written, so a digest's outcome is always
399    /// the last word on it.
400    fn settle_unsent(
401        &self,
402        pipeline: &Pipeline,
403        plan: &DeliveryPlan,
404        signal: Signal,
405        admission: &mut Admission,
406    ) -> DeliveryResult {
407        let result = summarize(plan, &[], None);
408        if let Err(e) = self.sink.complete(&signal, &result) {
409            tracing::error!("sink write failed: {e}");
410        }
411        self.hold(pipeline, plan, &signal, admission, true);
412        result
413    }

415    /// Adds `signal` to a group of each grouped route in `plan`, and
416    /// starts the timer of any group it opens.
417    fn hold(
418        &self,
419        pipeline: &Pipeline,
420        plan: &DeliveryPlan,
421        signal: &Signal,
422        admission: &mut Admission,
423        owns_row: bool,
424    ) {
425        for (index, _) in &plan.grouped {
426            let Some(route) = pipeline.routing.routes.get(*index) else {
427                continue;
428            };
429            let member = Member {
430                signal: signal.clone(),
431                admission: admission.split_off(route),
432                owns_row,
433            };
434            match self.groups.add(route, member) {
435                Added::Joined => {}
436                Added::Full(closed) => {
437                    self.spawn_group(closed);
438                }
439                Added::Opened { key, seq, wait } => {
440                    let groups = Arc::clone(&self.groups);
441                    let sender = self.group_sender();
442                    self.runtime.spawn(async move {
443                        tokio::time::sleep(wait).await;
444                        if let Some(closed) = groups.take(&key, seq) {
445                            sender.send(closed).await;
446                        }
447                    });
448                }
449            }
450        }
451    }

456    }

466    /// Blocking, bounded by timeout. Caller gets a definitive result.
467    /// Same pipeline as notify_async — only the blocking behavior
468    /// differs. On a timeout, providers that already delivered still
469    /// count, so a fan-out cut short reports `Partial` rather than
470    /// `Failed`.
471    ///
472    /// A signal that only grouped routes take returns `Grouped` at once:
473    /// it goes out in a digest after the route's `group_wait`, or on
474    /// [`SignalClient::flush_groups`] or [`SignalClient::flush`].

476    pub fn notify_sync(
477        &self,

479        timeout: Duration,

484    /// [`SignalClient::notify_sync`] that also returns the signal's id
485    /// and dedup key, for callers that need to refer to it afterwards.
486    /// The Python `send()` and C `signal_send` call this.
487    #[must_use = "check whether the signal was delivered"]
488    pub fn send(&self, input: SignalInput, timeout: Duration) -> Result<SendReport, SignalError> {
489        signal::validate(&input)?;
490        let pipeline = self.pipeline();
491        let Prepared {
492            signal,
493            plan,
494            mut admission,
495            prior_refs,
496        } = self.prepare(input, &pipeline);
497        record(self.sink.as_ref(), &signal);
498        if plan.targets.is_empty() {
499            let (signal_id, dedup_key) = (signal.id, signal.dedup_key.clone());
500            let result = self.settle_unsent(&pipeline, &plan, signal, &mut admission);
501            return Ok(SendReport {
502                signal_id,
503                dedup_key,
504                result,
505            });
506        }
507        self.hold(&pipeline, &plan, &signal, &mut admission, false);
508        let tracker = Tracker::new(&signal, self.sink.as_ref(), prior_refs);
509
510        let result = self.runtime.block_on(async {
511            match tokio::time::timeout(timeout, deliver(&tracker, &plan, &pipeline.providers)).await
512            {
513                Ok(result) => result,
514                Err(_) => summarize(
515                    &plan,
516                    &tracker.outcomes(),
517                    Some(&format!("timed out after {timeout:?}")),
518                ),
519            }
520        });
521        settle(self.throttle.as_deref(), &admission, &result);
522        if let Some(alerts) = &self.alerts {
523            alerts.record(&signal, &result, &tracker.refs());
524        }
525
526        self.sink.complete(&signal, &result)?;
527        Ok(SendReport {
528            signal_id: signal.id,
529            dedup_key: signal.dedup_key.clone(),
530            result,
531        })
532    }

534    /// Stamps a validated input and plans its delivery. A trigger goes
535    /// through the routes, held back by their `dedup` and `rate_limit`.
536    /// An `acknowledge` or `resolve` goes to the providers holding its
537    /// alert instead (see [`lifecycle_plan`]); routes don't throttle it,
538    /// since holding back a resolve would leave the alert open.
539    fn prepare(&self, mut input: SignalInput, pipeline: &Pipeline) -> Prepared {
540        if !input.action.is_lifecycle() {
541            let signal = Signal::from_input(input);
542            let (plan, admission) =
543                plan_signal(self.throttle.as_deref(), &pipeline.routing, &signal);
544            return Prepared {
545                signal,
546                plan,
547                admission,
548                prior_refs: HashMap::new(),
549            };
550        }
551        let open = match &self.alerts {
552            Some(alerts) => alerts.open_alerts(&input.effective_dedup_key()),
553            None => Vec::new(),
554        };
555        input.fill_lifecycle_defaults(open.first().map(|a| a.title.as_str()));
556        let signal = Signal::from_input(input);
557        let plan = lifecycle_plan(pipeline, &signal, &open);
558        for line in &plan.suppressed {
559            tracing::info!("signal {} skipped: {line}", signal.id);
560        }
561        let prior_refs = open
562            .into_iter()
563            .filter_map(|a| Some((a.provider, a.alert_ref?)))
564            .collect();
565        Prepared {
566            signal,
567            plan,
568            admission: Admission::default(),
569            prior_refs,
570        }
571    }
572}

574/// A signal ready to deliver.
575struct Prepared {
576    signal: Signal,
577    plan: DeliveryPlan,
578    admission: Admission,
579    /// Each provider's reference to the alert an `acknowledge` or
580    /// `resolve` acts on.
581    prior_refs: HashMap<String, String>,
582}
583
584/// Where an `acknowledge` or `resolve` goes: every provider holding its
585/// alert, each on its own. With no record of the alert (another host
586/// raised it, or it was resolved already), the providers that resolve
587/// by key on routes matching the signal whatever its severity; those
588/// ignore a key they have no open alert for. Message providers are left
589/// out then, so a success callback that runs after every task doesn't
590/// post "resolved" for alerts that never fired. With neither, nothing
591/// is sent and the signal is `skipped`.
592fn lifecycle_plan(pipeline: &Pipeline, signal: &Signal, open: &[OpenAlert]) -> DeliveryPlan {
593    let mut targets: Vec<Vec<String>> = open
594        .iter()
595        .filter(|a| pipeline.providers.contains_key(&a.provider))
596        .map(|a| vec![a.provider.clone()])
597        .collect();
598    if targets.is_empty() {
599        targets = pipeline
600            .routing
601            .providers_any_severity(signal)
602            .into_iter()
603            .filter(|name| {
604                pipeline
605                    .providers
606                    .get(name)
607                    .is_some_and(|p| p.resolves_by_key())
608            })
609            .map(|name| vec![name])
610            .collect();
611    }
612    let suppressed = if targets.is_empty() {
613        vec![format!(
614            "no open alert with dedup key {} to {}, and no route to a provider that resolves by key",
615            signal.dedup_key.as_deref().unwrap_or_default(),
616            signal.action
617        )]
618    } else {
619        Vec::new()
620    };
621    DeliveryPlan {
622        targets,
623        suppressed,
624        grouped: Vec::new(),
625    }
626}

646/// Plans a signal's delivery, holding back the routes whose `dedup` or
647/// `rate_limit` turn it down.
648fn plan_signal(
649    throttle: Option<&Throttle>,
650    routing: &RoutingConfig,
651    signal: &Signal,
652) -> (DeliveryPlan, Admission) {
653    let mut admission = Admission::default();
654    let now = throttle::now_ms();
655    let plan = routing.plan_with(signal, |route| match throttle {
656        Some(t) => t.admit(route, signal, now, &mut admission),
657        None => Ok(()),
658    });
659    for line in &plan.suppressed {
660        tracing::info!("signal {} suppressed on {line}", signal.id);
661    }
662    (plan, admission)
663}

665/// A signal that reached no provider must not count as sent, or its
666/// next occurrence would be suppressed as a duplicate of an alert
667/// nobody got.
668fn settle(throttle: Option<&Throttle>, admission: &Admission, result: &DeliveryResult) {
669    if let (Some(t), DeliveryStatus::Failed) = (throttle, result.status) {
670        t.release(admission);
671    }
672}

674impl GroupSender {
675    /// Sends a closed group through its route alone: as a digest, or
676    /// as the signal itself when the group holds just one. The route's
677    /// `rate_limit` is checked against digests here. Each member whose
678    /// stored row is waiting on the digest takes its outcome.
679    async fn send(self, closed: Closed) -> DeliveryResult {
680        let sink = self.sink.as_ref();
681        let route = &closed.route;
682        let single = closed.members.len() == 1;
683        let signal = if single {
684            closed.members[0].signal.clone()
685        } else {
686            let digest = group::digest(&closed);
687            record(sink, &digest);
688            digest
689        };
690
691        let mut admission = Admission::default();
692        let gate = match &self.throttle {
693            Some(t) => t.admit_digest(route, &signal, throttle::now_ms(), &mut admission),
694            None => Ok(()),
695        };
696        let plan = match gate {
697            Ok(()) => DeliveryPlan {
698                targets: route.targets(),
699                suppressed: Vec::new(),
700                grouped: Vec::new(),
701            },
702            Err(reason) => {
703                let line = format!("route {}: {reason}", route.describe());
704                tracing::info!("digest {} suppressed on {line}", signal.id);
705                DeliveryPlan {
706                    targets: Vec::new(),
707                    suppressed: vec![line],
708                    grouped: Vec::new(),
709                }
710            }
711        };
712
713        let pipeline = current(&self.pipeline);
714        let tracker = Tracker::new(&signal, sink, HashMap::new());
715        let result = deliver(&tracker, &plan, &pipeline.providers).await;
716        settle(self.throttle.as_deref(), &admission, &result);
717        for member in &closed.members {
718            settle(self.throttle.as_deref(), &member.admission, &result);
719        }
720
721        if !single {
722            if let Err(e) = sink.complete(&signal, &result) {
723                tracing::error!("sink write failed: {e}");
724            }
725        }
726        for member in closed.members.iter().filter(|m| m.owns_row) {
727            if let Err(e) = sink.complete(&member.signal, &result) {
728                tracing::error!("sink write failed: {e}");
729            }
730        }
731        result
732    }
733}

750/// The per-provider outcomes of one delivery. Each is written to the
751/// sink as soon as it changes, so a timeout or exit that cuts delivery
752/// short leaves the providers that already have the signal on record.
753struct Tracker<'a> {
754    signal: &'a Signal,
755    sink: &'a dyn SignalSink,
756    outcomes: Mutex<Vec<ProviderOutcome>>,
757    /// What each provider returned for the alert this signal acts on.
758    prior_refs: HashMap<String, String>,
759    /// What each provider returned for this delivery.
760    refs: Mutex<HashMap<String, String>>,
761}
762
763impl<'a> Tracker<'a> {
764    fn new(
765        signal: &'a Signal,
766        sink: &'a dyn SignalSink,
767        prior_refs: HashMap<String, String>,
768    ) -> Self {
769        Tracker {
770            signal,
771            sink,
772            outcomes: Mutex::default(),
773            prior_refs,
774            refs: Mutex::default(),
775        }
776    }
777
778    fn keep_ref(&self, provider: &str, alert_ref: String) {
779        self.refs
780            .lock()
781            .unwrap_or_else(|e| e.into_inner())
782            .insert(provider.to_string(), alert_ref);
783    }
784
785    fn refs(&self) -> HashMap<String, String> {
786        self.refs.lock().unwrap_or_else(|e| e.into_inner()).clone()
787    }
788
789    fn report(&self, outcome: ProviderOutcome) {
790        if let Err(e) = self.sink.provider_outcome(self.signal, &outcome) {
791            tracing::error!("sink write failed: {e}");
792        }
793        let mut outcomes = self.outcomes.lock().unwrap_or_else(|e| e.into_inner());
794        match outcomes.iter_mut().find(|o| o.provider == outcome.provider) {
795            Some(existing) => *existing = outcome,
796            None => outcomes.push(outcome),
797        }
798    }
799
800    fn outcomes(&self) -> Vec<ProviderOutcome> {
801        self.outcomes
802            .lock()
803            .unwrap_or_else(|e| e.into_inner())
804            .clone()
805    }
806}

808/// Delivers to every target of the plan at once. Targets never wait on
809/// each other, and each retries only its own providers, so a provider
810/// that delivered is never sent the signal again because another failed.
811async fn deliver(
812    tracker: &Tracker<'_>,
813    plan: &DeliveryPlan,
814    providers: &HashMap<String, Arc<dyn Provider>>,
815) -> DeliveryResult {
816    futures_util::future::join_all(
817        plan.targets
818            .iter()
819            .map(|chain| deliver_chain(tracker, chain, providers)),
820    )
821    .await;
822    summarize(plan, &tracker.outcomes(), None)
823}
824
825/// Tries a target's providers in order until one succeeds.
826async fn deliver_chain(
827    tracker: &Tracker<'_>,
828    chain: &[String],
829    providers: &HashMap<String, Arc<dyn Provider>>,
830) {
831    for name in chain {
832        let Some(provider) = providers.get(name) else {
833            tracing::warn!("route references unknown provider: {name}");
834            continue;
835        };
836        let outcome = |status, error, attempts| ProviderOutcome {
837            provider: name.clone(),
838            status,
839            error,
840            attempts,
841        };
842        tracker.report(outcome(DeliveryStatus::Pending, None, 0));
843        let mut last_error = None;
844        // Basic retry: exponential backoff up to MAX_RETRY_ATTEMPTS.
845        // v0.2 roadmap item: add jitter, make configurable per-route.
846        for attempt in 1..=MAX_RETRY_ATTEMPTS {
847            let prior = tracker.prior_refs.get(name).map(String::as_str);
848            match provider.send_with_ref(tracker.signal, prior).await {
849                Ok(alert_ref) => {
850                    if let Some(alert_ref) = alert_ref {
851                        tracker.keep_ref(name, alert_ref);
852                    }
853                    tracker.report(outcome(DeliveryStatus::Delivered, None, attempt));
854                    return;
855                }
856                Err(e) => {
857                    last_error = Some(e.to_string());
858                    if attempt < MAX_RETRY_ATTEMPTS {
859                        tokio::time::sleep(Duration::from_millis(
860                            INITIAL_BACKOFF_MS * 2u64.pow(attempt),
861                        ))
862                        .await;
863                    }
864                }
865            }
866        }
867        tracker.report(outcome(
868            DeliveryStatus::Failed,
869            last_error,
870            MAX_RETRY_ATTEMPTS,
871        ));
872    }
873}

875/// Folds per-provider outcomes into one result. `cut_off` is set when
876/// delivery was stopped early; providers still pending then count as
877/// failed with that message.
878fn summarize(
879    plan: &DeliveryPlan,
880    outcomes: &[ProviderOutcome],
881    cut_off: Option<&str>,
882) -> DeliveryResult {
883    let mut deliveries = Vec::new();
884    let mut delivered = Vec::new();
885    let mut errors = Vec::new();
886    let mut attempts = 0;
887    for chain in &plan.targets {
888        let mut tried: Vec<ProviderOutcome> = chain
889            .iter()
890            .filter_map(|name| outcomes.iter().find(|o| &o.provider == name).cloned())
891            .collect();
892        if let Some(msg) = cut_off {
893            for o in tried
894                .iter_mut()
895                .filter(|o| o.status == DeliveryStatus::Pending)
896            {
897                o.status = DeliveryStatus::Failed;
898                o.error = Some(msg.to_string());
899            }
900        }
901        // A chain stops at its first success, so the provider tried
902        // last decides how the target went.
903        match tried.last() {
904            Some(o) if o.status == DeliveryStatus::Delivered => {
905                delivered.push(o.provider.clone());
906                attempts = attempts.max(o.attempts);
907            }
908            Some(o) => {
909                let error = o.error.clone().unwrap_or_else(|| o.status.to_string());
910                errors.push(format!("{}: {error}", o.provider));
911                if plan.targets.len() == 1 {
912                    errors = vec![error];
913                }
914                attempts = attempts.max(o.attempts);
915            }
916            None => errors.push("no provider delivered successfully".to_string()),
917        }
918        deliveries.extend(tried);
919    }
920
921    let status = if plan.targets.is_empty() && !plan.grouped.is_empty() {
922        // Held for a digest; the digest's outcome comes later.
923        DeliveryStatus::Grouped
924    } else if plan.targets.is_empty() {
925        // Every matching route was suppressed; nothing was meant to go.
926        errors = plan.suppressed.clone();
927        DeliveryStatus::Skipped
928    } else if delivered.len() == plan.targets.len() {
929        DeliveryStatus::Delivered
930    } else if delivered.is_empty() {
931        DeliveryStatus::Failed
932    } else {
933        DeliveryStatus::Partial
934    };
935    DeliveryResult {
936        status,
937        provider: (!delivered.is_empty()).then(|| delivered.join(",")),
938        error: (!errors.is_empty()).then(|| errors.join("; ")),
939        attempts,
940        deliveries,
941        suppressed: plan.suppressed.clone(),
942        grouped: plan.grouped.iter().map(|(_, line)| line.clone()).collect(),
943    }
944}

1572            self.keyed
1573        }

1576            unreachable!("the core calls send_with_ref")
1577        }

1580            &self,

1589        }
