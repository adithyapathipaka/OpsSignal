6/// Severity is a closed enum, not a free string, so the routing engine
7/// can match on it exhaustively instead of doing string comparison.
8#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
9#[serde(rename_all = "lowercase")]
10pub enum Severity {
11    #[default]
12    Info,
13    Success,
14    Warning,
15    Error,
16    Critical,
17}
18
19impl std::fmt::Display for Severity {
20    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
21        f.write_str(match self {
22            Severity::Info => "info",
23            Severity::Success => "success",
24            Severity::Warning => "warning",
25            Severity::Error => "error",
26            Severity::Critical => "critical",
27        })
28    }
29}
30
31impl std::str::FromStr for Severity {
32    type Err = String;
33
34    fn from_str(s: &str) -> Result<Self, Self::Err> {
35        match s.to_lowercase().as_str() {
36            "info" => Ok(Severity::Info),
37            "success" => Ok(Severity::Success),
38            "warning" => Ok(Severity::Warning),
39            "error" => Ok(Severity::Error),
40            "critical" => Ok(Severity::Critical),
41            other => Err(format!(
42                "unknown severity {other:?}; expected one of: info, success, warning, error, critical"
43            )),
44        }
45    }
46}

48/// The generic signal model. `id` and `timestamp` are always stamped by
49/// the core (never trusted from the caller) so every downstream
50/// component — sink, dedup, retry tracking — has a stable identity and
51/// ordering guarantee regardless of which language binding produced it.
52#[derive(Debug, Clone, Serialize, Deserialize)]
53pub struct Signal {
54    pub id: Uuid,
55    pub source: String,
56    pub service: Option<String>,
57    pub component: Option<String>,
58    pub event_type: String,
59    pub severity: Severity,
60    pub title: String,
61    pub message: Option<String>,
62    pub environment: Option<String>,
63    pub team: Option<String>,
64    pub run_id: Option<String>,
65    pub correlation_id: Option<String>,
66    pub trace_id: Option<String>,
67    pub log_url: Option<String>,
68    #[serde(default = "default_metadata")]
69    pub metadata: serde_json::Value,
70    pub timestamp: DateTime<Utc>,
71    pub dedup_key: Option<String>,
72}
73
74fn default_metadata() -> serde_json::Value {
75    serde_json::json!({})
76}

78/// Builder input — what callers actually construct. Notably missing
79/// `id` and `timestamp`: those are not caller-settable.
80#[derive(Debug, Clone, Default, Serialize, Deserialize)]
81pub struct SignalInput {
82    pub source: String,
83    pub service: Option<String>,
84    pub component: Option<String>,
85    pub event_type: String,
86    pub severity: Severity,
87    pub title: String,
88    pub message: Option<String>,
89    pub environment: Option<String>,
90    pub team: Option<String>,
91    pub run_id: Option<String>,
92    pub correlation_id: Option<String>,
93    pub trace_id: Option<String>,
94    pub log_url: Option<String>,
95    pub metadata: Option<serde_json::Value>,
96    /// Caller may supply an explicit dedup key; if absent, core computes
97    /// one from (source, event_type, title, environment).
98    pub dedup_key: Option<String>,
99}
100
101impl Signal {
102    pub fn from_input(input: SignalInput) -> Self {
103        let timestamp = Utc::now();
104        let dedup_key = input
105            .dedup_key
106            .clone()
107            .unwrap_or_else(|| compute_dedup_key(&input));
108
109        Signal {
110            id: Uuid::new_v4(),
111            source: input.source,
112            service: input.service,
113            component: input.component,
114            event_type: input.event_type,
115            severity: input.severity,
116            title: input.title,
117            message: input.message,
118            environment: input.environment,
119            team: input.team,
120            run_id: input.run_id,
121            correlation_id: input.correlation_id,
122            trace_id: input.trace_id,
123            log_url: input.log_url,
124            metadata: input.metadata.unwrap_or_else(default_metadata),
125            timestamp,
126            dedup_key: Some(dedup_key),
127        }
128    }
129}

131fn compute_dedup_key(input: &SignalInput) -> String {
132    let mut hasher = Sha256::new();
133    hasher.update(input.source.as_bytes());
134    hasher.update(b"|");
135    hasher.update(input.event_type.as_bytes());
136    hasher.update(b"|");
137    hasher.update(input.title.as_bytes());
138    hasher.update(b"|");
139    hasher.update(input.environment.as_deref().unwrap_or("").as_bytes());
140    format!("{:x}", hasher.finalize())
141}

143#[derive(Debug, thiserror::Error)]
144#[non_exhaustive]
145pub enum SignalError {
146    #[error("validation failed: {0}")]
147    Validation(String),
148    #[error("config error: {0}")]
149    Config(String),
150    #[error("delivery failed: {0}")]
151    Delivery(String),
152    #[error("sink write failed: {0}")]
153    Sink(String),
154    #[error("timeout after {0:?}")]
155    Timeout(std::time::Duration),
156    #[error("internal error (recovered from panic): {0}")]
157    Internal(String),
158}
159
160pub fn validate(input: &SignalInput) -> Result<(), SignalError> {
161    if input.source.trim().is_empty() {
162        return Err(SignalError::Validation("source must not be empty".into()));
163    }
164    if input.event_type.trim().is_empty() {
165        return Err(SignalError::Validation(
166            "event_type must not be empty".into(),
167        ));
168    }
169    if input.title.trim().is_empty() {
170        return Err(SignalError::Validation("title must not be empty".into()));
171    }
172    Ok(())
173}
