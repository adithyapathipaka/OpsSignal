1//! Route `dedup` and `rate_limit`.
2//!
3//! The state lives in a `throttle` table of the SQLite file the sink
4//! writes to, not in memory, so it holds across every process on the
5//! host that shares that file: a CLI run from cron, each Airflow task,
6//! and a restarted service all see the same history. Coordinating hosts
7//! with each other is out of scope for the embedded SDK.
8//!
9//! Each admitted send is one row: the route, the signal's dedup key and
10//! the time. A route is keyed by a hash of its whole definition, so
11//! editing a route starts its history afresh.
12//!
13//! Every storage error lets the signal through. Throttling exists to
14//! cut noise; it must never be the reason an alert went missing.
15
16use crate::routing::{RateLimit, Route, Window};
17use crate::signal::{Signal, SignalError};
18use rusqlite::{Connection, OptionalExtension, TransactionBehavior};
19use sha2::{Digest, Sha256};
20use std::sync::Mutex;
21use std::time::Duration;
22
23/// How long a process waits on another one holding the file's write
24/// lock before giving up (and letting the signal through).
25const BUSY_TIMEOUT: Duration = Duration::from_secs(2);
26
27pub(crate) struct Throttle {
28    conn: Mutex<Connection>,
29}
30
31/// The sends recorded for one signal, so they can be taken back if
32/// delivery fails.
33#[derive(Debug, Default)]
34pub(crate) struct Admission {
35    rows: Vec<(String, String, i64)>,
36}
37
38impl Admission {
39    /// Moves out the sends recorded for `route`, so a route whose
40    /// delivery happens later (a digest) can release its own.
41    pub(crate) fn split_off(&mut self, route: &Route) -> Admission {
42        let key = route_key(route);
43        let (mine, rest) = std::mem::take(&mut self.rows)
44            .into_iter()
45            .partition(|(r, _, _)| *r == key);
46        self.rows = rest;
47        Admission { rows: mine }
48    }
49}
50
51impl Throttle {
52    pub(crate) fn open(path: &str) -> Result<Self, SignalError> {
53        let err = |e: rusqlite::Error| SignalError::Sink(format!("throttle store: {e}"));
54        let conn = Connection::open(path).map_err(err)?;
55        conn.busy_timeout(BUSY_TIMEOUT).map_err(err)?;
56        conn.execute_batch(
57            "CREATE TABLE IF NOT EXISTS throttle (
58                route TEXT NOT NULL,
59                dedup_key TEXT NOT NULL,
60                sent_at INTEGER NOT NULL
61            );
62            CREATE INDEX IF NOT EXISTS idx_throttle_route ON throttle(route, sent_at);",
63        )
64        .map_err(err)?;
65        Ok(Throttle {
66            conn: Mutex::new(conn),
67        })
68    }
69
70    /// Decides whether `route` may send `signal` at `now_ms` (Unix
71    /// milliseconds) and, if so, records the send in `admission`. For a
72    /// grouped route only `dedup` is checked here: its `rate_limit`
73    /// counts digests, through [`Throttle::admit_digest`].
74    pub(crate) fn admit(
75        &self,
76        route: &Route,
77        signal: &Signal,
78        now_ms: i64,
79        admission: &mut Admission,
80    ) -> Result<(), String> {
81        let rate = if route.is_grouped() {
82            None
83        } else {
84            route.rate_limit
85        };
86        self.check(
87            &route_key(route),
88            signal,
89            now_ms,
90            route.dedup,
91            rate,
92            admission,
93        )
94    }
95
96    /// Decides whether a grouped `route` may send the digest `signal`
97    /// under its `rate_limit`. Digests are counted apart from the
98    /// signals they hold, so the limit caps messages, not members.
99    pub(crate) fn admit_digest(
100        &self,
101        route: &Route,
102        signal: &Signal,
103        now_ms: i64,
104        admission: &mut Admission,
105    ) -> Result<(), String> {
106        let key = format!("{}#digest", route_key(route));
107        self.check(&key, signal, now_ms, None, route.rate_limit, admission)
108    }
109
110    fn check(
111        &self,
112        route_key: &str,
113        signal: &Signal,
114        now_ms: i64,
115        dedup: Option<Window>,
116        rate: Option<RateLimit>,
117        admission: &mut Admission,
118    ) -> Result<(), String> {
119        if dedup.is_none() && rate.is_none() {
120            return Ok(());
121        }
122        let key = signal.dedup_key.clone().unwrap_or_default();
123        match self.try_admit(route_key, &key, now_ms, dedup, rate) {
124            Ok(Ok(())) => {
125                admission.rows.push((route_key.to_string(), key, now_ms));
126                Ok(())
127            }
128            Ok(Err(reason)) => Err(reason),
129            Err(e) => {
130                tracing::error!("throttle check failed, sending anyway: {e}");
131                Ok(())
132            }
133        }
134    }
135
136    fn try_admit(
137        &self,
138        route_key: &str,
139        key: &str,
140        now_ms: i64,
141        dedup: Option<Window>,
142        rate: Option<RateLimit>,
143    ) -> rusqlite::Result<Result<(), String>> {
144        let mut conn = self.conn.lock().unwrap_or_else(|e| e.into_inner());
145        // IMMEDIATE takes the write lock up front, so two processes
146        // can't both see an empty window and both send.
147        let tx = conn.transaction_with_behavior(TransactionBehavior::Immediate)?;
148
149        if let Some(window) = dedup {
150            let last: Option<i64> = tx
151                .query_row(
152                    "SELECT MAX(sent_at) FROM throttle WHERE route = ?1 AND dedup_key = ?2",
153                    rusqlite::params![route_key, key],
154                    |r| r.get(0),
155                )
156                .optional()?
157                .flatten();
158            if let Some(last) = last.filter(|&t| now_ms - t < millis(window.0)) {
159                return Ok(Err(format!(
160                    "duplicate of a signal sent {} ago (dedup {window})",
161                    ago(now_ms - last)
162                )));
163            }
164        }
165        if let Some(limit) = rate {
166            let sent: i64 = tx.query_row(
167                "SELECT COUNT(*) FROM throttle WHERE route = ?1 AND sent_at > ?2",
168                rusqlite::params![route_key, now_ms - millis(limit.per.0)],
169                |r| r.get(0),
170            )?;
171            if sent >= i64::from(limit.max) {
172                return Ok(Err(format!(
173                    "rate limit of {} per {} reached",
174                    limit.max, limit.per
175                )));
176            }
177        }
178
179        tx.execute(
180            "INSERT INTO throttle (route, dedup_key, sent_at) VALUES (?1, ?2, ?3)",
181            rusqlite::params![route_key, key, now_ms],
182        )?;
183        // Rows older than both windows can no longer decide anything.
184        let keep = [dedup.map(|w| w.0), rate.map(|l| l.per.0)]
185            .into_iter()
186            .flatten()
187            .max()
188            .unwrap_or_default();
189        tx.execute(
190            "DELETE FROM throttle WHERE route = ?1 AND sent_at <= ?2",
191            rusqlite::params![route_key, now_ms - millis(keep)],
192        )?;
193        tx.commit()?;
194        Ok(Ok(()))
195    }
196
197    /// Forgets the sends of a signal that no provider delivered, so
198    /// the next occurrence of the incident is not suppressed as a
199    /// duplicate of an alert nobody received.
200    pub(crate) fn release(&self, admission: &Admission) {
201        if admission.rows.is_empty() {
202            return;
203        }
204        let conn = self.conn.lock().unwrap_or_else(|e| e.into_inner());
205        for (route, key, at) in &admission.rows {
206            if let Err(e) = conn.execute(
207                "DELETE FROM throttle WHERE rowid IN (SELECT rowid FROM throttle \
208                 WHERE route = ?1 AND dedup_key = ?2 AND sent_at = ?3 LIMIT 1)",
209                rusqlite::params![route, key, at],
210            ) {
211                tracing::error!("throttle release failed: {e}");
212            }
213        }
214    }
215}

217pub(crate) fn route_key(route: &Route) -> String {
218    let json = serde_json::to_string(route).unwrap_or_default();
219    format!("{:x}", Sha256::digest(json.as_bytes()))
220}
221
222fn millis(d: Duration) -> i64 {
223    i64::try_from(d.as_millis()).unwrap_or(i64::MAX)
224}
225
226fn ago(ms: i64) -> String {
227    let secs = ms.max(0) / 1000;
228    match secs {
229        0..=59 => format!("{secs}s"),
230        60..=3599 => format!("{}m", secs / 60),
231        _ => format!("{}h", secs / 3600),
232    }
233}
234
235pub(crate) fn now_ms() -> i64 {
236    chrono::Utc::now().timestamp_millis()
237}

245    const SEC: i64 = 1000;

247    fn route(dedup: Option<u64>, rate: Option<(u32, u64)>) -> Route {
248        Route {
249            match_rule: RouteMatch {
250                severity: vec![],
251                environment: None,
252                source: None,
253            },
254            providers: vec!["pagerduty".into()],
255            mode: DeliveryMode::Failover,
256            dedup: dedup.map(|s| Window(Duration::from_secs(s))),
257            rate_limit: rate.map(|(max, per)| RateLimit {
258                max,
259                per: Window(Duration::from_secs(per)),
260            }),
261            group_by: None,
262            group_wait: None,
263        }
264    }
265
266    fn signal(title: &str) -> Signal {
267        Signal::from_input(SignalInput {
268            source: "airflow".into(),
269            event_type: "task_failed".into(),
270            title: title.into(),
271            ..Default::default()
272        })
273    }
274
275    fn memory() -> Throttle {
276        Throttle::open(":memory:").expect("open")
277    }
278
279    fn admit(t: &Throttle, route: &Route, signal: &Signal, at: i64) -> Result<(), String> {
280        t.admit(route, signal, at, &mut Admission::default())
281    }

283    #[test]
284    fn dedup_suppresses_repeats_inside_the_window_only() {
285        let t = memory();
286        let r = route(Some(600), None);
287        let s = signal("disk full");
288        assert!(admit(&t, &r, &s, 0).is_ok());
289        let err = admit(&t, &r, &s, 120 * SEC).expect_err("duplicate");
290        assert_eq!(err, "duplicate of a signal sent 2m ago (dedup 10m)");
291        assert!(admit(&t, &r, &s, 599 * SEC).is_err());
292        // The window counts from the last send, not the last repeat, so
293        // a long incident re-alerts once per window.
294        assert!(admit(&t, &r, &s, 600 * SEC).is_ok());
295    }

316    #[test]
317    fn rate_limit_is_a_sliding_window_across_keys() {
318        let t = memory();
319        let r = route(None, Some((2, 60)));
320        assert!(admit(&t, &r, &signal("a"), 0).is_ok());
321        assert!(admit(&t, &r, &signal("b"), 10 * SEC).is_ok());
322        let err = admit(&t, &r, &signal("c"), 20 * SEC).expect_err("limited");
323        assert_eq!(err, "rate limit of 2 per 1m reached");
324        // The first send leaves the window after 60s.
325        assert!(admit(&t, &r, &signal("c"), 61 * SEC).is_ok());
326        assert!(admit(&t, &r, &signal("d"), 62 * SEC).is_err());
327    }

352    #[test]
353    fn history_is_shared_through_the_file() {
354        // Two stores on one file stand in for two processes.
355        let db = std::env::temp_dir()
356            .join(format!("opssignal-throttle-{}.db", uuid::Uuid::new_v4()))
357            .to_string_lossy()
358            .into_owned();
359        let r = route(Some(600), None);
360        let s = signal("a");
361        assert!(admit(&Throttle::open(&db).expect("open"), &r, &s, 0).is_ok());
362        assert!(admit(&Throttle::open(&db).expect("open"), &r, &s, SEC).is_err());
363        let _ = std::fs::remove_file(&db);
364    }
