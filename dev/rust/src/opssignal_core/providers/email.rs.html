11/// Covers connect, TLS, auth and the whole SMTP exchange of one attempt,
12/// so a server that hangs costs one attempt rather than the retry loop.
13const SMTP_TIMEOUT: Duration = Duration::from_secs(30);

15/// Sends a plain-text email through an SMTP server or, in a build with
16/// AWS support, the Amazon SES API. Repeats of the same
17/// signal keep the same subject and `References` header, so mail
18/// clients thread them into one conversation instead of filling the
19/// inbox.
20pub struct EmailProvider {
21    config: EmailConfig,
22    templates: Option<Templates>,
23}

31    /// `subject_prefix` goes before a templated subject too, so mail
32    /// filters keep working. A subject is one line.
33    fn subject_for(&self, signal: &Signal) -> String {
34        let subject = match template::render(&self.templates, "subject", signal, &[]) {
35            Some(text) => text.split_whitespace().collect::<Vec<_>>().join(" "),
36            None => {
37                let severity = signal.severity.to_string().to_uppercase();
38                format!("[{severity}] {}", signal.title)
39            }
40        };
41        match &self.config.subject_prefix {
42            Some(prefix) if !prefix.is_empty() => format!("{prefix} {subject}"),
43            _ => subject,
44        }
45    }
46
47    fn body_for(&self, signal: &Signal) -> String {
48        template::render(&self.templates, "body", signal, &[])
49            .unwrap_or_else(|| Self::default_body_for(signal))
50    }
51
52    fn default_body_for(signal: &Signal) -> String {
53        let mut body = format!("{}\n", signal.title);
54        if let Some(message) = &signal.message {
55            body.push('\n');
56            body.push_str(message);
57            body.push('\n');
58        }
59        body.push('\n');
60        let severity = signal.severity.to_string();
61        let mut fields: Vec<(&str, &str)> = vec![
62            ("Severity", &severity),
63            ("Source", &signal.source),
64            ("Event", &signal.event_type),
65        ];
66        let optional = [
67            ("Environment", &signal.environment),
68            ("Service", &signal.service),
69            ("Component", &signal.component),
70            ("Team", &signal.team),
71            ("Run ID", &signal.run_id),
72            ("Correlation ID", &signal.correlation_id),
73            ("Trace ID", &signal.trace_id),
74            ("Logs", &signal.log_url),
75        ];
76        for (label, value) in optional {
77            if let Some(value) = value {
78                fields.push((label, value));
79            }
80        }
81        for (label, value) in fields {
82            body.push_str(&format!("{:<16}{value}\n", format!("{label}:")));
83        }
84        if signal.metadata.as_object().is_some_and(|m| !m.is_empty()) {
85            let metadata = serde_json::to_string_pretty(&signal.metadata)
86                .unwrap_or_else(|_| signal.metadata.to_string());
87            body.push_str(&format!("\nMetadata:\n{metadata}\n"));
88        }
89        body.push_str(&format!(
90            "\n--\nSignal {} at {}\nSent by OpsSignal\n",
91            signal.id,
92            signal.timestamp.to_rfc3339()
93        ));
94        body
95    }

97    fn message_for(&self, signal: &Signal) -> Result<Message, SignalError> {
98        let mut builder: MessageBuilder = Message::builder()
99            .from(mailbox(&self.config.from)?)
100            .subject(self.subject_for(signal))
101            .header(ContentType::TEXT_PLAIN);
102        for to in &self.config.to {
103            builder = builder.to(mailbox(to)?);
104        }
105        for cc in &self.config.cc {
106            builder = builder.cc(mailbox(cc)?);
107        }
108        if let Some(key) = &signal.dedup_key {
109            let thread = thread_id(key);
110            builder = builder.in_reply_to(thread.clone()).references(thread);
111        }
112        builder
113            .body(self.body_for(signal))
114            .map_err(|e| SignalError::Delivery(format!("cannot build the email: {e}")))
115    }
116
117    fn transport(smtp: &SmtpConfig) -> Result<AsyncSmtpTransport<Tokio1Executor>, SignalError> {
118        let builder = match smtp.tls {
119            SmtpTls::Starttls => AsyncSmtpTransport::<Tokio1Executor>::starttls_relay(&smtp.host),
120            SmtpTls::Tls => AsyncSmtpTransport::<Tokio1Executor>::relay(&smtp.host),
121            SmtpTls::None => Ok(AsyncSmtpTransport::<Tokio1Executor>::builder_dangerous(
122                &smtp.host,
123            )),
124        }
125        .map_err(|e| SignalError::Delivery(format!("smtp: {e}")))?;
126        let mut builder = builder.port(smtp.port()).timeout(Some(SMTP_TIMEOUT));
127        if let (Some(username), Some(password)) = (&smtp.username, &smtp.password) {
128            builder = builder.credentials(Credentials::new(username.clone(), password.clone()));
129        }
130        Ok(builder.build())
131    }
132}
133
134fn mailbox(address: &str) -> Result<Mailbox, SignalError> {
135    address
136        .parse()
137        .map_err(|e| SignalError::Delivery(format!("invalid email address {address:?}: {e}")))
138}

140/// A message id that every email for the same dedup key refers to.
141/// Hashed because caller-supplied keys may hold characters a header
142/// can't carry.
143fn thread_id(dedup_key: &str) -> String {
144    let digest = format!("{:x}", Sha256::digest(dedup_key.as_bytes()));
145    format!("<{}@opssignal>", &digest[..32])
146}

149impl Provider for EmailProvider {
150    fn name(&self) -> &'static str {
151        "email"
152    }
153
154    fn preview(&self, signal: &Signal) -> Option<serde_json::Value> {
155        Some(serde_json::json!({
156            "subject": self.subject_for(signal),
157            "body": self.body_for(signal),
158        }))
159    }
160
161    async fn send(&self, signal: &Signal) -> Result<(), SignalError> {
162        match self.config.transport() {
163            Some(EmailTransport::Smtp(smtp)) => {
164                let message = self.message_for(signal)?;
165                Self::transport(smtp)?
166                    .send(message)
167                    .await
168                    .map_err(|e| SignalError::Delivery(format!("smtp: {e}")))?;
169                Ok(())
170            }
171            Some(EmailTransport::Ses(ses)) => self.send_ses(ses, signal).await,
172            None => Err(SignalError::Delivery(
173                "email needs exactly one of smtp and ses".into(),
174            )),
175        }
176    }
177}

183    async fn send_ses(&self, ses: &SesConfig, signal: &Signal) -> Result<(), SignalError> {
184        let raw = self.message_for(signal)?.formatted();
185        crate::cloud::aws::send_ses_raw(
186            ses.region.as_deref(),
187            ses.configuration_set.as_deref(),
188            &raw,
189        )
190        .await
191    }

243    /// A one-connection SMTP server that accepts everything except
244    /// recipients in `reject_rcpt`, and records what the client sent.
245    async fn fake_smtp(reject_rcpt: bool) -> (u16, Arc<Mutex<String>>) {
246        let listener = TcpListener::bind("127.0.0.1:0").await.expect("bind");
247        let port = listener.local_addr().expect("addr").port();
248        let transcript = Arc::new(Mutex::new(String::new()));
249        let log = Arc::clone(&transcript);
250        tokio::spawn(async move {
251            let Ok((socket, _)) = listener.accept().await else {
252                return;
253            };
254            let (read, mut write) = socket.into_split();
255            let mut lines = BufReader::new(read).lines();
256            let _ = write.write_all(b"220 fake ESMTP\r\n").await;
257            let mut in_data = false;
258            while let Ok(Some(line)) = lines.next_line().await {
259                log.lock().expect("log").push_str(&format!("{line}\n"));
260                let reply: &[u8] = if in_data {
261                    if line == "." {
262                        in_data = false;
263                        b"250 queued\r\n"
264                    } else {
265                        continue;
266                    }
267                } else {
268                    let verb = line.split(' ').next().unwrap_or("").to_uppercase();
269                    match verb.as_str() {
270                        "EHLO" | "HELO" => b"250-fake\r\n250 AUTH PLAIN LOGIN\r\n",
271                        "AUTH" => b"235 ok\r\n",
272                        "RCPT" if reject_rcpt => b"550 no such user\r\n",
273                        "DATA" => {
274                            in_data = true;
275                            b"354 go ahead\r\n"
276                        }
277                        "QUIT" => {
278                            let _ = write.write_all(b"221 bye\r\n").await;
279                            return;
280                        }
281                        _ => b"250 ok\r\n",
282                    }
283                };
284                let _ = write.write_all(reply).await;
285            }
286        });
287        (port, transcript)
288    }
