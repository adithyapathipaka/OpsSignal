5/// Events API v2 endpoint for US-hosted accounts. EU accounts use
6/// `https://events.eu.pagerduty.com/v2/enqueue`, set through `url`.
7pub const DEFAULT_EVENTS_URL: &str = "https://events.pagerduty.com/v2/enqueue";
8
9/// PagerDuty rejects longer values with a 400, which retrying cannot
10/// fix, so they are cut to fit instead.
11const MAX_SUMMARY_CHARS: usize = 1024;
12const MAX_DEDUP_KEY_CHARS: usize = 255;
13
14/// Triggers a PagerDuty alert through the Events API v2. The signal's
15/// `dedup_key` becomes PagerDuty's, so a repeat of the same signal adds
16/// to the open alert instead of paging again.
17pub struct PagerDutyProvider {
18    routing_key: String,
19    pub url: String,
20    client: reqwest::Client,
21    templates: Option<Templates>,
22}
23
24impl PagerDutyProvider {
25    pub fn new(routing_key: String, url: Option<String>) -> Self {
26        PagerDutyProvider {
27            routing_key,
28            url: url.unwrap_or_else(|| DEFAULT_EVENTS_URL.to_string()),
29            client: reqwest::Client::new(),
30            templates: None,
31        }
32    }
33
34    /// Uses `templates` for the parts they set; see [`crate::template`].
35    pub fn with_templates(mut self, templates: Option<Templates>) -> Self {
36        self.templates = templates;
37        self
38    }
39
40    /// PagerDuty has no `success` severity; a success signal is
41    /// informational.
42    fn severity_for(severity: Severity) -> &'static str {
43        match severity {
44            Severity::Info | Severity::Success => "info",
45            Severity::Warning => "warning",
46            Severity::Error => "error",
47            Severity::Critical => "critical",
48        }
49    }
50
51    fn event_for(&self, signal: &Signal) -> serde_json::Value {
52        let mut details = serde_json::Map::new();
53        details.insert("signal_id".into(), signal.id.to_string().into());
54        details.insert("event_type".into(), signal.event_type.clone().into());
55        details.insert("severity".into(), signal.severity.to_string().into());
56        let optional = [
57            ("message", &signal.message),
58            ("environment", &signal.environment),
59            ("team", &signal.team),
60            ("run_id", &signal.run_id),
61            ("correlation_id", &signal.correlation_id),
62            ("trace_id", &signal.trace_id),
63        ];
64        for (key, value) in optional {
65            if let Some(value) = value {
66                details.insert(key.into(), value.clone().into());
67            }
68        }
69        if signal.metadata.as_object().is_some_and(|m| !m.is_empty()) {
70            details.insert("metadata".into(), signal.metadata.clone());
71        }
72
73        let summary = template::render(&self.templates, "summary", signal, &[])
74            .unwrap_or_else(|| signal.title.clone());
75        let mut payload = serde_json::json!({
76            "summary": truncate(&summary, MAX_SUMMARY_CHARS),
77            "source": signal.source,
78            "severity": Self::severity_for(signal.severity),
79            "timestamp": signal.timestamp.to_rfc3339(),
80            "class": signal.event_type,
81            "custom_details": details,
82        });
83        if let Some(component) = &signal.component {
84            payload["component"] = component.clone().into();
85        }
86        if let Some(service) = &signal.service {
87            payload["group"] = service.clone().into();
88        }
89
90        let mut event = serde_json::json!({
91            "routing_key": self.routing_key,
92            "event_action": "trigger",
93            "payload": payload,
94        });
95        if let Some(key) = &signal.dedup_key {
96            event["dedup_key"] = truncate(key, MAX_DEDUP_KEY_CHARS).into();
97        }
98        if let Some(log_url) = &signal.log_url {
99            event["links"] = serde_json::json!([{ "href": log_url, "text": "Logs" }]);
100        }
101        event
102    }
103}
104
105fn truncate(s: &str, max_chars: usize) -> String {
106    match s.char_indices().nth(max_chars) {
107        Some((end, _)) => s[..end].to_string(),
108        None => s.to_string(),
109    }
110}

112#[async_trait::async_trait]
113impl Provider for PagerDutyProvider {
114    fn name(&self) -> &'static str {
115        "pagerduty"
116    }
117
118    /// The routing key is a credential, so it is left out.
119    fn preview(&self, signal: &Signal) -> Option<serde_json::Value> {
120        let mut event = self.event_for(signal);
121        event["routing_key"] = "<redacted>".into();
122        Some(event)
123    }
124
125    async fn send(&self, signal: &Signal) -> Result<(), SignalError> {
126        let resp = self
127            .client
128            .post(&self.url)
129            .json(&self.event_for(signal))
130            .send()
131            .await
132            .map_err(|e| SignalError::Delivery(e.to_string()))?;
133
134        if !resp.status().is_success() {
135            return Err(SignalError::Delivery(format!(
136                "pagerduty events API returned status {}",
137                resp.status()
138            )));
139        }
140        Ok(())
141    }
142}
