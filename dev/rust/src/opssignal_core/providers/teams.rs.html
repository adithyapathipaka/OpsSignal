4/// Posts an Adaptive Card to a Microsoft Teams channel webhook. The
5/// message shape works with both Workflows (Power Automate) webhooks,
6/// which replace Office 365 connectors, and legacy incoming webhooks.
7pub struct TeamsProvider {
8    pub webhook_url: String,
9    client: reqwest::Client,
10}
11
12impl TeamsProvider {
13    pub fn new(webhook_url: String) -> Self {
14        TeamsProvider {
15            webhook_url,
16            client: reqwest::Client::new(),
17        }
18    }

20    /// Adaptive Card text colours.
21    fn color_for(severity: Severity) -> &'static str {
22        match severity {
23            Severity::Info => "default",
24            Severity::Success => "good",
25            Severity::Warning => "warning",
26            Severity::Error | Severity::Critical => "attention",
27        }
28    }
29
30    fn card_for(signal: &Signal) -> serde_json::Value {
31        let mut body = vec![serde_json::json!({
32            "type": "TextBlock",
33            "text": signal.title,
34            "weight": "bolder",
35            "size": "medium",
36            "color": Self::color_for(signal.severity),
37            "wrap": true,
38        })];
39        if let Some(message) = &signal.message {
40            body.push(serde_json::json!({
41                "type": "TextBlock",
42                "text": message,
43                "wrap": true,
44            }));
45        }
46
47        let mut facts = vec![
48            fact("Severity", &signal.severity.to_string()),
49            fact("Source", &signal.source),
50            fact("Event", &signal.event_type),
51        ];
52        let optional = [
53            ("Environment", &signal.environment),
54            ("Service", &signal.service),
55            ("Component", &signal.component),
56            ("Team", &signal.team),
57            ("Run ID", &signal.run_id),
58        ];
59        for (title, value) in optional {
60            if let Some(value) = value {
61                facts.push(fact(title, value));
62            }
63        }
64        body.push(serde_json::json!({ "type": "FactSet", "facts": facts }));
65
66        let mut card = serde_json::json!({
67            "$schema": "http://adaptivecards.io/schemas/adaptive-card.json",
68            "type": "AdaptiveCard",
69            "version": "1.4",
70            "body": body,
71        });
72        if let Some(log_url) = &signal.log_url {
73            card["actions"] = serde_json::json!([{
74                "type": "Action.OpenUrl",
75                "title": "View logs",
76                "url": log_url,
77            }]);
78        }
79
80        serde_json::json!({
81            "type": "message",
82            "attachments": [{
83                "contentType": "application/vnd.microsoft.card.adaptive",
84                "contentUrl": null,
85                "content": card,
86            }],
87        })
88    }
89}

91fn fact(title: &str, value: &str) -> serde_json::Value {
92    serde_json::json!({ "title": title, "value": value })
93}

95#[async_trait::async_trait]
96impl Provider for TeamsProvider {
97    fn name(&self) -> &'static str {
98        "teams"
99    }
100
101    async fn send(&self, signal: &Signal) -> Result<(), SignalError> {
102        let resp = self
103            .client
104            .post(&self.webhook_url)
105            .json(&Self::card_for(signal))
106            .send()
107            .await
108            .map_err(|e| SignalError::Delivery(e.to_string()))?;
109
110        if !resp.status().is_success() {
111            return Err(SignalError::Delivery(format!(
112                "teams webhook returned status {}",
113                resp.status()
114            )));
115        }
116        Ok(())
117    }
118}

140    fn facts(card: &serde_json::Value) -> Vec<(String, String)> {
141        card["attachments"][0]["content"]["body"]
142            .as_array()
143            .unwrap()
144            .iter()
145            .find(|b| b["type"] == "FactSet")
146            .unwrap()["facts"]
147            .as_array()
148            .unwrap()
149            .iter()
150            .map(|f| {
151                (
152                    f["title"].as_str().unwrap().to_string(),
153                    f["value"].as_str().unwrap().to_string(),
154                )
155            })
156            .collect()
157    }

170        let card = TeamsProvider::card_for(&signal(Severity::Error));
171        assert_eq!(card["type"], "message");
172        let attachment = &card["attachments"][0];
173        assert_eq!(

175            "application/vnd.microsoft.card.adaptive"

177        let content = &attachment["content"];

180        assert_eq!(content["body"][0]["color"], "attention");
