1//! `signal.yaml` loading.
2//!
3//! The file is resolved from `$SIGNAL_CONFIG` first (a local path, or an
4//! `https://`, `s3://`, `gs://` or `az://` URL; see [`crate::source`]), then
5//! `./signal.yaml`.
6//! If neither exists the SDK still works: it falls back to the
7//! zero-config default (console provider, SQLite sink at
8//! `./signals.db`), so `notify()` never needs setup to do something
9//! useful.
10//!
11//! `${VAR}` placeholders in string values are replaced from the
12//! environment so secrets like Slack webhook URLs stay out of the file.
13//! A placeholder whose variable is unset is a config error, not an
14//! empty string: a blank webhook URL would only fail later, at delivery
15//! time, which is exactly when nobody is looking.
16

40/// One file of a layered config: what it sets itself, plus the files
41/// it builds on. [`Config::from_layers`] merges a chain of these.
42#[derive(Debug, Clone, Default, Deserialize)]
43#[serde(deny_unknown_fields)]
44pub struct Layer {
45    /// Parent files, as URLs or paths relative to this file. Each
46    /// parent is merged in before this file, so this file wins.
47    #[serde(default, deserialize_with = "one_or_many")]
48    pub extends: Vec<String>,
49    #[serde(default)]
50    pub providers: ProvidersConfig,
51    #[serde(default)]
52    pub routes: Vec<Route>,
53    #[serde(default)]
54    pub storage: Option<StorageConfig>,
55}

57fn one_or_many<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Vec<String>, D::Error> {
58    #[derive(Deserialize)]
59    #[serde(untagged)]
60    enum OneOrMany {
61        One(String),
62        Many(Vec<String>),
63    }
64    Ok(match OneOrMany::deserialize(d)? {
65        OneOrMany::One(s) => vec![s],
66        OneOrMany::Many(v) => v,
67    })
68}

70impl Layer {
71    /// Parses one file, interpolating `${VAR}` placeholders. Routes are
72    /// checked only once the whole chain is merged, since a team file
73    /// may route to a provider its parent configures.
74    ///
75    /// A `${store:...}` secret placeholder is an error here, since it
76    /// needs a network read; configs loaded through a client resolve
77    /// them first (see [`crate::secrets`]).
78    pub fn from_yaml_str(text: &str) -> Result<Self, SignalError> {
79        Self::from_yaml_str_with(text, &HashMap::new())
80    }
81
82    /// Like [`Layer::from_yaml_str`], taking secret placeholders from
83    /// `secrets` (keyed by placeholder name, such as `aws-sm:prod/x`).
84    pub(crate) fn from_yaml_str_with(
85        text: &str,
86        secrets: &HashMap<String, String>,
87    ) -> Result<Self, SignalError> {
88        let mut value = parse_value(text)?;
89        interpolate(&mut value, secrets)?;
90        serde_yaml::from_value(value).map_err(|e| config_err(e.to_string()))
91    }
92
93    /// The names of the secret placeholders in `text`, each once, in
94    /// order. Placeholders in YAML comments are not included.
95    pub(crate) fn secret_refs(text: &str) -> Result<Vec<String>, SignalError> {
96        fn walk(value: &serde_yaml::Value, out: &mut Vec<String>) -> Result<(), SignalError> {
97            match value {
98                serde_yaml::Value::String(s) => {
99                    for name in placeholders(s)? {
100                        if crate::secrets::is_ref(name) && !out.iter().any(|n| n == name) {
101                            out.push(name.to_string());
102                        }
103                    }
104                }
105                serde_yaml::Value::Sequence(items) => {
106                    for item in items {
107                        walk(item, out)?;
108                    }
109                }
110                serde_yaml::Value::Mapping(map) => {
111                    for (_, v) in map {
112                        walk(v, out)?;
113                    }
114                }
115                serde_yaml::Value::Tagged(tagged) => walk(&tagged.value, out)?,
116                _ => {}
117            }
118            Ok(())
119        }
120        let mut out = Vec::new();
121        walk(&parse_value(text)?, &mut out)?;
122        Ok(out)
123    }
124}
125
126fn parse_value(text: &str) -> Result<serde_yaml::Value, SignalError> {
127    let value: serde_yaml::Value =
128        serde_yaml::from_str(text).map_err(|e| config_err(format!("invalid YAML: {e}")))?;
129    // An empty file parses as null; treat it as "all defaults".
130    Ok(if value.is_null() {
131        serde_yaml::Value::Mapping(Default::default())
132    } else {
133        value
134    })
135}

161impl ProvidersConfig {
162    /// Compiles every provider's templates once, so a mistake in one is
163    /// a load-time error rather than a fallback at send time.
164    fn check_templates(&self) -> Result<(), SignalError> {
165        let all: [(&'static str, Option<&TemplateConfig>); 7] = [
166            ("console", self.console.as_ref().map(|c| &c.template)),
167            ("slack", self.slack.as_ref().map(|c| &c.template)),
168            ("webhook", self.webhook.as_ref().map(|c| &c.template)),
169            ("pagerduty", self.pagerduty.as_ref().map(|c| &c.template)),
170            ("teams", self.teams.as_ref().map(|c| &c.template)),
171            ("opsgenie", self.opsgenie.as_ref().map(|c| &c.template)),
172            ("email", self.email.as_ref().map(|c| &c.template)),
173        ];
174        for (name, template) in all {
175            if let Some(template) = template {
176                Templates::compile(name, template)?;
177            }
178        }
179        Ok(())
180    }

293impl SlackConfig {
294    fn check(&self) -> Result<(), SignalError> {
295        match (&self.webhook_url, &self.bot_token, &self.channel) {
296            (Some(_), None, None) | (None, Some(_), Some(_)) => Ok(()),
297            (Some(_), Some(_), _) => Err(config_err(
298                "providers.slack: set webhook_url or bot_token, not both",
299            )),
300            (None, Some(_), None) => Err(config_err(
301                "providers.slack.bot_token needs a channel to post to",
302            )),
303            (Some(_), None, Some(_)) => Err(config_err(
304                "providers.slack.channel is used only with bot_token; a webhook posts to its own channel",
305            )),
306            (None, None, _) => Err(config_err(
307                "providers.slack needs webhook_url or bot_token",
308            )),
309        }
310    }

313/// How an [`EmailConfig`] sends.
314pub enum EmailTransport<'a> {
315    Smtp(&'a SmtpConfig),
316    Ses(&'a SesConfig),
317}
318
319impl EmailConfig {
320    /// The transport; [`EmailConfig::check`] makes sure there is
321    /// exactly one.
322    pub fn transport(&self) -> Option<EmailTransport<'_>> {
323        match (&self.smtp, &self.ses) {
324            (Some(smtp), None) => Some(EmailTransport::Smtp(smtp)),
325            (None, Some(ses)) => Some(EmailTransport::Ses(ses)),
326            _ => None,
327        }
328    }
329
330    /// Addresses are checked at load time, since a bad one would fail
331    /// every send.
332    fn check(&self) -> Result<(), SignalError> {
333        if self.to.is_empty() {
334            return Err(config_err("providers.email.to lists no recipients"));
335        }
336        let addresses = std::iter::once(&self.from).chain(&self.to).chain(&self.cc);
337        for address in addresses {
338            address.parse::<lettre::message::Mailbox>().map_err(|e| {
339                config_err(format!("providers.email: invalid address {address:?}: {e}"))
340            })?;
341        }
342        match self.transport() {
343            Some(EmailTransport::Smtp(smtp)) => {
344                if smtp.username.is_some() != smtp.password.is_some() {
345                    return Err(config_err(
346                        "providers.email.smtp needs both username and password, or neither",
347                    ));
348                }
349            }
350            Some(EmailTransport::Ses(_)) => {
351                if !crate::providers::external::cloud_supported("aws") {
352                    return Err(config_err(format!(
353                        "providers.email: {}",
354                        crate::providers::external::missing_support("email over Amazon SES", "aws")
355                    )));
356                }
357            }
358            None => {
359                return Err(config_err(
360                    "providers.email needs exactly one of smtp and ses",
361                ))
362            }
363        }
364        Ok(())
365    }
366}

447impl Config {
448    /// Parses a single, self-contained config from YAML text,
449    /// interpolating `${VAR}` placeholders and checking that every route
450    /// names a configured provider. A file that uses `extends` needs to
451    /// be loaded from where it lives, with [`Config::from_file`] or
452    /// `$SIGNAL_CONFIG`, so its parents can be found.
453    pub fn from_yaml_str(text: &str) -> Result<Self, SignalError> {
454        let layer = Layer::from_yaml_str(text)?;
455        if !layer.extends.is_empty() {
456            return Err(config_err(
457                "`extends` needs the file's location; load it with Config::from_file or $SIGNAL_CONFIG",
458            ));
459        }
460        Self::from_layers([(None, layer)])
461    }
462
463    /// Merges a chain of layers, given lowest priority first, each with
464    /// the name of the file it came from (for error messages):
465    ///
466    /// - providers merge by name, and a later layer's entry replaces an
467    ///   earlier one's;
468    /// - a later layer's routes go before the earlier layers' routes,
469    ///   so the more specific file is matched first and the shared
470    ///   defaults still catch everything else;
471    /// - `storage` comes from the last layer that sets it.
472    ///
473    /// Routes are checked against the merged providers.
474    pub fn from_layers(
475        layers: impl IntoIterator<Item = (Option<String>, Layer)>,
476    ) -> Result<Self, SignalError> {
477        let mut config = Config::default();
478        // (file, index within that file) for each merged route.
479        let mut origins: Vec<(Option<String>, usize)> = Vec::new();
480        for (name, layer) in layers {
481            let p = layer.providers;
482            config.providers.console = p.console.or(config.providers.console.take());
483            config.providers.slack = p.slack.or(config.providers.slack.take());
484            config.providers.webhook = p.webhook.or(config.providers.webhook.take());
485            config.providers.pagerduty = p.pagerduty.or(config.providers.pagerduty.take());
486            config.providers.teams = p.teams.or(config.providers.teams.take());
487            config.providers.opsgenie = p.opsgenie.or(config.providers.opsgenie.take());
488            config.providers.email = p.email.or(config.providers.email.take());
489            if let Some(storage) = layer.storage {
490                config.storage = storage;
491            }
492            let mut routes = layer.routes;
493            let mut route_origins: Vec<_> = (0..routes.len()).map(|i| (name.clone(), i)).collect();
494            routes.append(&mut config.routes);
495            route_origins.append(&mut origins);
496            config.routes = routes;
497            origins = route_origins;
498        }
499        if let Some(email) = &config.providers.email {
500            email.check()?;
501        }
502        if let Some(slack) = &config.providers.slack {
503            slack.check()?;
504        }
505        config.providers.check_templates()?;
506        config.check_routes(&origins)?;
507        Ok(config)
508    }
509
510    /// Loads and merges `sources` once, resolving secret placeholders;
511    /// see [`crate::source`].
512    pub fn load(sources: &[ConfigSource], opts: &LoadOptions) -> Result<Self, SignalError> {
513        source::load_blocking(sources, opts).map(|l| l.config)
514    }
515
516    pub fn from_file(path: &Path) -> Result<Self, SignalError> {
517        let source = ConfigSource::File(path.to_path_buf());
518        source::load_blocking(std::slice::from_ref(&source), &LoadOptions::once()).map(|l| l.config)
519    }
520
521    /// `$SIGNAL_CONFIG` if set (one or more comma-separated local
522    /// files, which must then exist, or remote URLs), else
523    /// `./signal.yaml` if present, else the zero-config default. Remote
524    /// files are fetched once, with the cache fallback described in
525    /// [`crate::source`].
526    pub fn discover() -> Result<Self, SignalError> {
527        match ConfigSource::discover()? {
528            Some(sources) => {
529                source::load_blocking(&sources, &LoadOptions::from_env()?).map(|l| l.config)
530            }
531            None => Ok(Config::default()),
532        }
533    }

541    /// Names of the providers this config makes available. `console`
542    /// is always present because routing falls back to it.
543    pub fn provider_names(&self) -> Vec<&'static str> {
544        let mut names = vec!["console"];
545        if self.providers.slack.is_some() {
546            names.push("slack");
547        }
548        if self.providers.webhook.is_some() {
549            names.push("webhook");
550        }
551        if self.providers.pagerduty.is_some() {
552            names.push("pagerduty");
553        }
554        if self.providers.teams.is_some() {
555            names.push("teams");
556        }
557        if self.providers.opsgenie.is_some() {
558            names.push("opsgenie");
559        }
560        if self.providers.email.is_some() {
561            names.push("email");
562        }
563        names
564    }
565
566    fn check_routes(&self, origins: &[(Option<String>, usize)]) -> Result<(), SignalError> {
567        let known = self.provider_names();
568        for (route, (file, i)) in self.routes.iter().zip(origins) {
569            let at = match file {
570                Some(file) => format!("{file}: routes[{i}]"),
571                None => format!("routes[{i}]"),
572            };
573            if route.providers.is_empty() {
574                return Err(config_err(format!("{at} lists no providers")));
575            }
576            if route.dedup.is_some_and(|w| w.0.is_zero()) {
577                return Err(config_err(format!("{at}: dedup must be longer than 0s")));
578            }
579            if let Some(limit) = route.rate_limit {
580                if limit.max == 0 || limit.per.0.is_zero() {
581                    return Err(config_err(format!(
582                        "{at}: rate_limit needs max and per above 0"
583                    )));
584                }
585            }
586            if route.group_wait.is_some_and(|w| w.0.is_zero()) {
587                return Err(config_err(format!(
588                    "{at}: group_wait must be longer than 0s"
589                )));
590            }
591            for field in route.group_by.iter().flatten() {
592                if !crate::group::is_group_field(field) {
593                    return Err(config_err(format!(
594                        "{at}: cannot group_by '{field}' (use one of {}, or metadata.<key>)",
595                        crate::routing::GROUP_FIELDS.join(", ")
596                    )));
597                }
598            }
599            for name in &route.providers {
600                if !known.contains(&name.as_str()) {
601                    return Err(config_err(format!(
602                        "{at} references provider '{name}', which is not configured under providers (configured: {})",
603                        known.join(", ")
604                    )));
605                }
606            }
607        }
608        Ok(())
609    }
610}

616fn interpolate(
617    value: &mut serde_yaml::Value,
618    secrets: &HashMap<String, String>,
619) -> Result<(), SignalError> {
620    match value {
621        serde_yaml::Value::String(s) => {
622            *s = interpolate_str(s, secrets)?;
623        }
624        serde_yaml::Value::Sequence(items) => {
625            for item in items {
626                interpolate(item, secrets)?;
627            }
628        }
629        serde_yaml::Value::Mapping(map) => {
630            for (_, v) in map.iter_mut() {
631                interpolate(v, secrets)?;
632            }
633        }
634        serde_yaml::Value::Tagged(tagged) => interpolate(&mut tagged.value, secrets)?,
635        _ => {}
636    }
637    Ok(())
638}

640/// The `${...}` names in `input`, in order.
641fn placeholders(input: &str) -> Result<Vec<&str>, SignalError> {
642    let mut names = Vec::new();
643    let mut rest = input;
644    while let Some(start) = rest.find("${") {
645        let after = &rest[start + 2..];
646        let end = after
647            .find('}')
648            .ok_or_else(|| config_err(format!("unterminated ${{...}} in '{input}'")))?;
649        let name = &after[..end];
650        if name.is_empty() {
651            return Err(config_err(format!("empty ${{}} placeholder in '{input}'")));
652        }
653        names.push(name);
654        rest = &after[end + 1..];
655    }
656    Ok(names)
657}
658
659fn interpolate_str(input: &str, secrets: &HashMap<String, String>) -> Result<String, SignalError> {
660    let mut out = String::with_capacity(input.len());
661    let mut rest = input;
662    for name in placeholders(input)? {
663        let placeholder = format!("${{{name}}}");
664        let start = rest.find(&placeholder).expect("found by placeholders");
665        out.push_str(&rest[..start]);
666        let val = if crate::secrets::is_ref(name) {
667            secrets.get(name).cloned().ok_or_else(|| {
668                config_err(format!(
669                    "secret ${{{name}}} can only be read when the config is loaded by a client"
670                ))
671            })?
672        } else {
673            std::env::var(name).map_err(|_| {
674                config_err(format!(
675                    "environment variable {name} is referenced in signal.yaml but not set"
676                ))
677            })?
678        };
679        out.push_str(&val);
680        rest = &rest[start + placeholder.len()..];
681    }
682    out.push_str(rest);
683    Ok(out)
684}
