1//! `signal.yaml` loading.
2//!
3//! The file is resolved from `$SIGNAL_CONFIG` first (a local path, or an
4//! `https://`, `s3://`, `gs://` or `az://` URL; see [`crate::source`]), then
5//! `./signal.yaml`.
6//! If neither exists the SDK still works: it falls back to the
7//! zero-config default (console provider, SQLite sink at
8//! `./signals.db`), so `notify()` never needs setup to do something
9//! useful.
10//!
11//! `${VAR}` placeholders in string values are replaced from the
12//! environment so secrets like Slack webhook URLs stay out of the file.
13//! A placeholder whose variable is unset is a config error, not an
14//! empty string: a blank webhook URL would only fail later, at delivery
15//! time, which is exactly when nobody is looking.
16

41/// it builds on. [`Config::from_layers`] merges a chain of these.

44pub struct Layer {

46    /// parent is merged in before this file, so this file wins.

48    pub extends: Vec<String>,
49    #[serde(default)]

57fn one_or_many<'de, D: serde::Deserializer<'de>>(d: D) -> Result<Vec<String>, D::Error> {
58    #[derive(Deserialize)]
59    #[serde(untagged)]
60    enum OneOrMany {
61        One(String),
62        Many(Vec<String>),
63    }
64    Ok(match OneOrMany::deserialize(d)? {
65        OneOrMany::One(s) => vec![s],
66        OneOrMany::Many(v) => v,
67    })
68}

70impl Layer {
71    /// Parses one file, interpolating `${VAR}` placeholders. Routes are
72    /// checked only once the whole chain is merged, since a team file
73    /// may route to a provider its parent configures.
74    ///
75    /// A `${store:...}` secret placeholder is an error here, since it
76    /// needs a network read; configs loaded through a client resolve
77    /// them first (see [`crate::secrets`]).
78    pub fn from_yaml_str(text: &str) -> Result<Self, SignalError> {
79        Self::from_yaml_str_with(text, &HashMap::new())
80    }
81
82    /// Like [`Layer::from_yaml_str`], taking secret placeholders from
83    /// `secrets` (keyed by placeholder name, such as `aws-sm:prod/x`).
84    pub(crate) fn from_yaml_str_with(
85        text: &str,
86        secrets: &HashMap<String, String>,
87    ) -> Result<Self, SignalError> {
88        let mut value = parse_value(text)?;
89        interpolate(&mut value, secrets)?;
90        serde_yaml::from_value(value).map_err(|e| config_err(e.to_string()))
91    }
92
93    /// The names of the secret placeholders in `text`, each once, in
94    /// order. Placeholders in YAML comments are not included.
95    pub(crate) fn secret_refs(text: &str) -> Result<Vec<String>, SignalError> {
96        fn walk(value: &serde_yaml::Value, out: &mut Vec<String>) -> Result<(), SignalError> {
97            match value {
98                serde_yaml::Value::String(s) => {
99                    for name in placeholders(s)? {
100                        if crate::secrets::is_ref(name) && !out.iter().any(|n| n == name) {
101                            out.push(name.to_string());
102                        }
103                    }
104                }
105                serde_yaml::Value::Sequence(items) => {
106                    for item in items {
107                        walk(item, out)?;
108                    }
109                }
110                serde_yaml::Value::Mapping(map) => {
111                    for (_, v) in map {
112                        walk(v, out)?;
113                    }
114                }
115                serde_yaml::Value::Tagged(tagged) => walk(&tagged.value, out)?,
116                _ => {}
117            }
118            Ok(())
119        }
120        let mut out = Vec::new();
121        walk(&parse_value(text)?, &mut out)?;
122        Ok(out)
123    }
124}
125
126fn parse_value(text: &str) -> Result<serde_yaml::Value, SignalError> {
127    let value: serde_yaml::Value =
128        serde_yaml::from_str(text).map_err(|e| config_err(format!("invalid YAML: {e}")))?;
129    // An empty file parses as null; treat it as "all defaults".
130    Ok(if value.is_null() {
131        serde_yaml::Value::Mapping(Default::default())
132    } else {
133        value
134    })
135}

161impl ProvidersConfig {
162    /// Compiles every provider's templates once, so a mistake in one is
163    /// a load-time error rather than a fallback at send time.
164    fn check_templates(&self) -> Result<(), SignalError> {
165        let all: [(&'static str, Option<&TemplateConfig>); 7] = [
166            ("console", self.console.as_ref().map(|c| &c.template)),
167            ("slack", self.slack.as_ref().map(|c| &c.template)),
168            ("webhook", self.webhook.as_ref().map(|c| &c.template)),
169            ("pagerduty", self.pagerduty.as_ref().map(|c| &c.template)),
170            ("teams", self.teams.as_ref().map(|c| &c.template)),
171            ("opsgenie", self.opsgenie.as_ref().map(|c| &c.template)),
172            ("email", self.email.as_ref().map(|c| &c.template)),
173        ];
174        for (name, template) in all {
175            if let Some(template) = template {
176                Templates::compile(name, template)?;
177            }
178        }
179        Ok(())
180    }

279/// How an [`EmailConfig`] sends.
280pub enum EmailTransport<'a> {
281    Smtp(&'a SmtpConfig),
282    Ses(&'a SesConfig),
283}
284
285impl EmailConfig {
286    /// The transport; [`EmailConfig::check`] makes sure there is
287    /// exactly one.
288    pub fn transport(&self) -> Option<EmailTransport<'_>> {
289        match (&self.smtp, &self.ses) {
290            (Some(smtp), None) => Some(EmailTransport::Smtp(smtp)),
291            (None, Some(ses)) => Some(EmailTransport::Ses(ses)),
292            _ => None,
293        }
294    }
295
296    /// Addresses are checked at load time, since a bad one would fail
297    /// every send.
298    fn check(&self) -> Result<(), SignalError> {
299        if self.to.is_empty() {
300            return Err(config_err("providers.email.to lists no recipients"));
301        }
302        let addresses = std::iter::once(&self.from).chain(&self.to).chain(&self.cc);
303        for address in addresses {
304            address.parse::<lettre::message::Mailbox>().map_err(|e| {
305                config_err(format!("providers.email: invalid address {address:?}: {e}"))
306            })?;
307        }
308        match self.transport() {
309            Some(EmailTransport::Smtp(smtp)) => {
310                if smtp.username.is_some() != smtp.password.is_some() {
311                    return Err(config_err(
312                        "providers.email.smtp needs both username and password, or neither",
313                    ));
314                }
315            }
316            Some(EmailTransport::Ses(_)) => {
317                if !crate::providers::external::cloud_supported("aws") {
318                    return Err(config_err(format!(
319                        "providers.email: {}",
320                        crate::providers::external::missing_support("email over Amazon SES", "aws")
321                    )));
322                }
323            }
324            None => {
325                return Err(config_err(
326                    "providers.email needs exactly one of smtp and ses",
327                ))
328            }
329        }
330        Ok(())
331    }
332}

413impl Config {
414    /// Parses a single, self-contained config from YAML text,
415    /// interpolating `${VAR}` placeholders and checking that every route
416    /// names a configured provider. A file that uses `extends` needs to
417    /// be loaded from where it lives, with [`Config::from_file`] or
418    /// `$SIGNAL_CONFIG`, so its parents can be found.
419    pub fn from_yaml_str(text: &str) -> Result<Self, SignalError> {
420        let layer = Layer::from_yaml_str(text)?;
421        if !layer.extends.is_empty() {
422            return Err(config_err(
423                "`extends` needs the file's location; load it with Config::from_file or $SIGNAL_CONFIG",
424            ));
425        }
426        Self::from_layers([(None, layer)])
427    }
428
429    /// Merges a chain of layers, given lowest priority first, each with
430    /// the name of the file it came from (for error messages):
431    ///
432    /// - providers merge by name, and a later layer's entry replaces an
433    ///   earlier one's;
434    /// - a later layer's routes go before the earlier layers' routes,
435    ///   so the more specific file is matched first and the shared
436    ///   defaults still catch everything else;
437    /// - `storage` comes from the last layer that sets it.
438    ///
439    /// Routes are checked against the merged providers.
440    pub fn from_layers(
441        layers: impl IntoIterator<Item = (Option<String>, Layer)>,
442    ) -> Result<Self, SignalError> {
443        let mut config = Config::default();
444        // (file, index within that file) for each merged route.
445        let mut origins: Vec<(Option<String>, usize)> = Vec::new();
446        for (name, layer) in layers {
447            let p = layer.providers;
448            config.providers.console = p.console.or(config.providers.console.take());
449            config.providers.slack = p.slack.or(config.providers.slack.take());
450            config.providers.webhook = p.webhook.or(config.providers.webhook.take());
451            config.providers.pagerduty = p.pagerduty.or(config.providers.pagerduty.take());
452            config.providers.teams = p.teams.or(config.providers.teams.take());
453            config.providers.opsgenie = p.opsgenie.or(config.providers.opsgenie.take());
454            config.providers.email = p.email.or(config.providers.email.take());
455            if let Some(storage) = layer.storage {
456                config.storage = storage;
457            }
458            let mut routes = layer.routes;
459            let mut route_origins: Vec<_> = (0..routes.len()).map(|i| (name.clone(), i)).collect();
460            routes.append(&mut config.routes);
461            route_origins.append(&mut origins);
462            config.routes = routes;
463            origins = route_origins;
464        }
465        if let Some(email) = &config.providers.email {
466            email.check()?;
467        }
468        config.providers.check_templates()?;
469        config.check_routes(&origins)?;
470        Ok(config)
471    }

473    /// Loads and merges `sources` once, resolving secret placeholders;
474    /// see [`crate::source`].
475    pub fn load(sources: &[ConfigSource], opts: &LoadOptions) -> Result<Self, SignalError> {
476        source::load_blocking(sources, opts).map(|l| l.config)
477    }

479    pub fn from_file(path: &Path) -> Result<Self, SignalError> {
480        let source = ConfigSource::File(path.to_path_buf());
481        source::load_blocking(std::slice::from_ref(&source), &LoadOptions::once()).map(|l| l.config)
482    }

484    /// `$SIGNAL_CONFIG` if set (one or more comma-separated local
485    /// files, which must then exist, or remote URLs), else
486    /// `./signal.yaml` if present, else the zero-config default. Remote
487    /// files are fetched once, with the cache fallback described in
488    /// [`crate::source`].
489    pub fn discover() -> Result<Self, SignalError> {
490        match ConfigSource::discover()? {
491            Some(sources) => {
492                source::load_blocking(&sources, &LoadOptions::from_env()?).map(|l| l.config)
493            }
494            None => Ok(Config::default()),
495        }
496    }

504    /// Names of the providers this config makes available. `console`
505    /// is always present because routing falls back to it.
506    pub fn provider_names(&self) -> Vec<&'static str> {
507        let mut names = vec!["console"];
508        if self.providers.slack.is_some() {
509            names.push("slack");
510        }
511        if self.providers.webhook.is_some() {
512            names.push("webhook");
513        }
514        if self.providers.pagerduty.is_some() {
515            names.push("pagerduty");
516        }
517        if self.providers.teams.is_some() {
518            names.push("teams");
519        }
520        if self.providers.opsgenie.is_some() {
521            names.push("opsgenie");
522        }
523        if self.providers.email.is_some() {
524            names.push("email");
525        }
526        names
527    }
528
529    fn check_routes(&self, origins: &[(Option<String>, usize)]) -> Result<(), SignalError> {
530        let known = self.provider_names();
531        for (route, (file, i)) in self.routes.iter().zip(origins) {
532            let at = match file {
533                Some(file) => format!("{file}: routes[{i}]"),
534                None => format!("routes[{i}]"),
535            };
536            if route.providers.is_empty() {
537                return Err(config_err(format!("{at} lists no providers")));
538            }
539            for name in &route.providers {
540                if !known.contains(&name.as_str()) {
541                    return Err(config_err(format!(
542                        "{at} references provider '{name}', which is not configured under providers (configured: {})",
543                        known.join(", ")
544                    )));
545                }
546            }
547        }
548        Ok(())
549    }
550}

552fn config_err(msg: impl Into<String>) -> SignalError {

556fn interpolate(
557    value: &mut serde_yaml::Value,
558    secrets: &HashMap<String, String>,
559) -> Result<(), SignalError> {
560    match value {
561        serde_yaml::Value::String(s) => {
562            *s = interpolate_str(s, secrets)?;
563        }
564        serde_yaml::Value::Sequence(items) => {
565            for item in items {
566                interpolate(item, secrets)?;
567            }
568        }
569        serde_yaml::Value::Mapping(map) => {
570            for (_, v) in map.iter_mut() {
571                interpolate(v, secrets)?;
572            }
573        }
574        serde_yaml::Value::Tagged(tagged) => interpolate(&mut tagged.value, secrets)?,
575        _ => {}
576    }
577    Ok(())
578}

580/// The `${...}` names in `input`, in order.
581fn placeholders(input: &str) -> Result<Vec<&str>, SignalError> {
582    let mut names = Vec::new();
583    let mut rest = input;
584    while let Some(start) = rest.find("${") {
585        let after = &rest[start + 2..];
586        let end = after
587            .find('}')
588            .ok_or_else(|| config_err(format!("unterminated ${{...}} in '{input}'")))?;
589        let name = &after[..end];
590        if name.is_empty() {
591            return Err(config_err(format!("empty ${{}} placeholder in '{input}'")));
592        }
593        names.push(name);
594        rest = &after[end + 1..];
595    }
596    Ok(names)
597}
598
599fn interpolate_str(input: &str, secrets: &HashMap<String, String>) -> Result<String, SignalError> {
600    let mut out = String::with_capacity(input.len());
601    let mut rest = input;
602    for name in placeholders(input)? {
603        let placeholder = format!("${{{name}}}");
604        let start = rest.find(&placeholder).expect("found by placeholders");
605        out.push_str(&rest[..start]);
606        let val = if crate::secrets::is_ref(name) {
607            secrets.get(name).cloned().ok_or_else(|| {
608                config_err(format!(
609                    "secret ${{{name}}} can only be read when the config is loaded by a client"
610                ))
611            })?
612        } else {
613            std::env::var(name).map_err(|_| {
614                config_err(format!(
615                    "environment variable {name} is referenced in signal.yaml but not set"
616                ))
617            })?
618        };
619        out.push_str(&val);
620        rest = &rest[start + placeholder.len()..];
621    }
622    out.push_str(rest);
623    Ok(out)
624}
